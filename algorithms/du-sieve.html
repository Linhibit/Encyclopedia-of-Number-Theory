杜教筛 (Du Sieve)

算法原理：

杜教筛是一种用于计算数论函数前缀和的算法，由中国算法竞赛选手杜瑜皓在2016年提出。其核心思想是利用狄利克雷卷积的性质，通过递归和记忆化技术，在 $O(n^{2/3})$ 时间复杂度内计算积性函数的前缀和。

核心公式： 设 $f$ 和 $g$ 是两个数论函数，$h = f * g$ 是它们的狄利克雷卷积，即： $$h(n) = \sum_{d|n} f(d) g\left(\frac{n}{d}\right)$$ 记 $S_f(n) = \sum_{i=1}^n f(i)$，则有： $$\sum_{i=1}^n h(i) = \sum_{i=1}^n \sum_{d|i} f(d) g\left(\frac{i}{d}\right) = \sum_{d=1}^n f(d) \sum_{k=1}^{\lfloor n/d \rfloor} g(k)$$ 即： $$\sum_{i=1}^n h(i) = \sum_{d=1}^n f(d) S_g\left(\lfloor \frac{n}{d} \rfloor\right)$$

整理得杜教筛递推公式： $$g(1) S_f(n) = \sum_{i=1}^n h(i) - \sum_{d=2}^n f(d) S_g\left(\lfloor \frac{n}{d} \rfloor\right)$$

算法步骤：

  1. 输入：
    • 目标函数 $f(n)$（积性函数）
    • 辅助函数 $g(n)$，满足 $g(1) \neq 0$
    • 卷积函数 $h(n) = (f * g)(n)$，且 $h(n)$ 的前缀和容易计算
    • 正整数 $n$
  2. 预处理：
    1. 线性筛预处理 $f$ 的前 $m = \lfloor n^{2/3} \rfloor$ 个值，计算 $S_f(1)$ 到 $S_f(m)$
    2. 同样预处理 $g$ 和 $h$ 的前 $m$ 个值及其前缀和
  3. 递归计算： 对于给定的 $n$，计算 $S_f(n)$：
    1. 如果 $n \leq m$，直接返回预计算的 $S_f(n)$
    2. 如果 $n > m$，检查是否已计算过 $S_f(n)$（记忆化）
    3. 如果未计算，使用杜教筛公式： $$S_f(n) = \frac{1}{g(1)} \left( S_h(n) - \sum_{d=2}^n f(d) S_g\left(\lfloor \frac{n}{d} \rfloor\right) \right)$$
    4. 其中求和通过整数分块加速：$\lfloor n/d \rfloor$ 只有 $O(\sqrt{n})$ 个不同的值
    5. 递归计算需要的 $S_g(\lfloor n/d \rfloor)$ 值
  4. 优化技巧：
    • 使用哈希表记忆化已计算的 $S_f$ 值
    • 预处理前 $n^{2/3}$ 项以平衡时间和空间
    • 选择合适的辅助函数 $g$ 使得 $h$ 的前缀和容易计算
复杂度分析：
  • 预处理：$O(n^{2/3})$ 时间和空间
  • 递归计算：$O(n^{2/3})$ 时间（主定理分析）
  • 总复杂度：$O(n^{2/3})$，优于直接计算的 $O(n)$ 和朴素分块的 $O(\sqrt{n})$ 递归深度
  • 适用于 $n$ 高达 $10^{10}$ 甚至更大的情况

经典应用：

例1：计算欧拉函数 $\varphi(n)$ 的前缀和
设 $f(n) = \varphi(n)$，取 $g(n) = 1$（常值函数），则： $$h(n) = (\varphi * 1)(n) = \sum_{d|n} \varphi(d) = n$$ 因此 $S_h(n) = \sum_{i=1}^n i = \frac{n(n+1)}{2}$。
杜教筛公式为： $$S_\varphi(n) = \frac{n(n+1)}{2} - \sum_{d=2}^n S_\varphi\left(\lfloor \frac{n}{d} \rfloor\right)$$

例2：计算莫比乌斯函数 $\mu(n)$ 的前缀和
设 $f(n) = \mu(n)$，取 $g(n) = 1$，则： $$h(n) = (\mu * 1)(n) = \varepsilon(n) = \begin{cases} 1 & n=1 \\ 0 & n>1 \end{cases}$$ 因此 $S_h(n) = 1$。
杜教筛公式为： $$S_\mu(n) = 1 - \sum_{d=2}^n S_\mu\left(\lfloor \frac{n}{d} \rfloor\right)$$

例3：计算约数个数函数 $d(n)$ 的前缀和
设 $f(n) = d(n)$，取 $g(n) = \mu(n)$，则： $$h(n) = (d * \mu)(n) = 1$$ 因为 $d = 1 * 1$，所以 $d * \mu = 1 * 1 * \mu = 1 * \varepsilon = 1$。
因此 $S_h(n) = n$。
杜教筛公式为： $$S_d(n) = n - \sum_{d=2}^n \mu(d) S_d\left(\lfloor \frac{n}{d} \rfloor\right)$$ 但这里需要 $\mu$ 的前缀和，可以同时计算。

代码示例 (JavaScript)：

// 线性筛预处理多个积性函数
function linearSieve(limit) {
    const primes = [];
    const isPrime = new Array(limit + 1).fill(true);
    const mu = new Array(limit + 1).fill(0);      // 莫比乌斯函数
    const phi = new Array(limit + 1).fill(0);     // 欧拉函数
    const d = new Array(limit + 1).fill(0);       // 约数个数
    const minPrime = new Array(limit + 1).fill(0); // 最小质因子
    
    // 初始化
    mu[1] = 1;
    phi[1] = 1;
    d[1] = 1;
    
    for (let i = 2; i <= limit; i++) {
        if (isPrime[i]) {
            primes.push(i);
            mu[i] = -1;
            phi[i] = i - 1;
            d[i] = 2;
            minPrime[i] = i;
        }
        
        for (const p of primes) {
            if (i * p > limit) break;
            
            const num = i * p;
            isPrime[num] = false;
            minPrime[num] = p;
            
            if (i % p === 0) {
                // i 包含质因子 p
                mu[num] = 0;
                phi[num] = phi[i] * p;
                
                // 计算 d(num)：如果 i = p^a * m，则 num = p^(a+1) * m
                let temp = i, a = 0;
                while (temp % p === 0) {
                    temp /= p;
                    a++;
                }
                d[num] = d[i] / (a + 1) * (a + 2);
                break;
            } else {
                // i 和 p 互质
                mu[num] = -mu[i];
                phi[num] = phi[i] * (p - 1);
                d[num] = d[i] * 2;
            }
        }
    }
    
    // 计算前缀和
    const sumMu = new Array(limit + 1).fill(0);
    const sumPhi = new Array(limit + 1).fill(0);
    const sumD = new Array(limit + 1).fill(0);
    
    for (let i = 1; i <= limit; i++) {
        sumMu[i] = sumMu[i - 1] + mu[i];
        sumPhi[i] = sumPhi[i - 1] + phi[i];
        sumD[i] = sumD[i - 1] + d[i];
    }
    
    return {
        primes, mu, phi, d,
        sumMu, sumPhi, sumD,
        minPrime
    };
}

// 杜教筛类
class DuSieve {
    constructor(n) {
        this.n = n;
        this.limit = Math.floor(Math.pow(n, 2/3)); // 预处理大小
        this.pre = linearSieve(this.limit);
        
        // 记忆化存储
        this.muMemo = new Map();
        this.phiMemo = new Map();
        this.dMemo = new Map();
    }
    
    // 计算莫比乌斯函数前缀和 Sμ(n)
    sumMuDu(n) {
        if (n <= this.limit) return this.pre.sumMu[n];
        if (this.muMemo.has(n)) return this.muMemo.get(n);
        
        let result = 1; // S_h(n) = 1 (h = μ * 1 = ε)
        
        // 整数分块求和
        for (let l = 2, r; l <= n; l = r + 1) {
            r = Math.floor(n / Math.floor(n / l));
            result -= (r - l + 1) * this.sumMuDu(Math.floor(n / l));
        }
        
        this.muMemo.set(n, result);
        return result;
    }
    
    // 计算欧拉函数前缀和 Sφ(n)
    sumPhiDu(n) {
        if (n <= this.limit) return this.pre.sumPhi[n];
        if (this.phiMemo.has(n)) return this.phiMemo.get(n);
        
        // S_h(n) = n(n+1)/2 (h = φ * 1 = id)
        let result = n * (n + 1) / 2;
        
        // 整数分块求和
        for (let l = 2, r; l <= n; l = r + 1) {
            r = Math.floor(n / Math.floor(n / l));
            result -= (r - l + 1) * this.sumPhiDu(Math.floor(n / l));
        }
        
        this.phiMemo.set(n, result);
        return result;
    }
    
    // 计算约数个数前缀和 Sd(n) - 需要同时计算 Sμ
    sumDDu(n) {
        if (n <= this.limit) return this.pre.sumD[n];
        if (this.dMemo.has(n)) return this.dMemo.get(n);
        
        // h = d * μ = 1, 所以 S_h(n) = n
        let result = n;
        
        // 需要 μ 的前缀和
        for (let l = 2, r; l <= n; l = r + 1) {
            r = Math.floor(n / Math.floor(n / l));
            const sumMuRange = this.sumMuDu(r) - this.sumMuDu(l - 1);
            result -= sumMuRange * this.sumDDu(Math.floor(n / l));
        }
        
        this.dMemo.set(n, result);
        return result;
    }
    
    // 通用的杜教筛模板
    // f: 目标函数，g: 辅助函数，h: 卷积函数，calcHSum: 计算h前缀和的函数
    duSieveGeneral(n, fPre, gPre, calcHSum) {
        if (n <= this.limit) return fPre[n];
        
        const memoKey = `f_${n}`;
        if (this.generalMemo && this.generalMemo.has(memoKey)) {
            return this.generalMemo.get(memoKey);
        }
        
        let result = calcHSum(n);
        
        // 整数分块
        for (let l = 2, r; l <= n; l = r + 1) {
            r = Math.floor(n / Math.floor(n / l));
            // 需要计算 ∑ g(d) for d in [l, r]
            // 这里假设我们可以快速计算g的前缀和
            const sumGRange = gPre[r] - gPre[l - 1];
            result -= sumGRange * this.duSieveGeneral(
                Math.floor(n / l), fPre, gPre, calcHSum
            );
        }
        
        // 除以 g(1)
        result /= gPre[1] - gPre[0];
        
        if (this.generalMemo) {
            this.generalMemo.set(memoKey, result);
        }
        
        return result;
    }
}

应用：

  1. 数论函数求和： 高效计算各种积性函数的前缀和，如 $\varphi(n)$、$\mu(n)$、$d(n)$、$\sigma(n)$ 等。
  2. 数论问题求解： 许多数论问题最终归结为计算某个数论函数的前缀和，杜教筛提供高效算法。
  3. 算法竞赛： 在要求计算大范围前缀和的题目中，杜教筛是常用工具。
  4. 数学研究： 在研究数论函数的渐近行为时，需要计算大量前缀和作为实验数据。
  5. 密码学： 在某些基于数论的密码算法分析中，需要计算相关函数的前缀和。

📝 优化与扩展：

1. 预处理大小选择： 通常取 $m = n^{2/3}$ 以达到理论最优复杂度 $O(n^{2/3})$。实际中可根据具体问题调整。
2. 记忆化优化： 使用哈希表存储已计算的结果，避免重复计算。
3. 多函数同时计算： 当需要计算多个相关函数的前缀和时，可以共享预处理和记忆化结果。
4. 递归深度控制： 通过预处理减少递归深度，提高实际运行效率。
5. 扩展性： 杜教筛可以推广到非积性函数，只要能找到合适的辅助函数 $g$ 使得 $h = f * g$ 的前缀和容易计算。
6. 局限性： 需要找到合适的辅助函数 $g$，且 $g$ 和 $h$ 的前缀和需要容易计算。对于某些函数可能不容易找到。

历史背景： 杜教筛由中国算法竞赛选手杜瑜皓在2016年提出，并在算法竞赛社区中广泛传播和应用。 它是对传统数论函数求和方法的重大改进，使得计算高达 $10^{10}$ 甚至更大的前缀和成为可能。
