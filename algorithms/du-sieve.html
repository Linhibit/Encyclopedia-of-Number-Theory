杜教筛 (Du Sieve)

算法原理：

杜教筛是一种用于计算数论函数前缀和的算法，由杜瑜皓在2016年提出。其核心思想是利用狄利克雷卷积的性质，通过递归和记忆化技术，在 $O(n^{2/3})$ 时间复杂度内计算积性函数的前缀和。

核心公式： 设 $f$ 和 $g$ 是两个数论函数，$h = f * g$ 是它们的狄利克雷卷积，即： $$h(n) = \sum_{d|n} f(d) g\left(\frac{n}{d}\right)$$ 记 $S_f(n) = \sum_{i=1}^n f(i)$，则有： $$\sum_{i=1}^n h(i) = \sum_{i=1}^n \sum_{d|i} f(d) g\left(\frac{i}{d}\right) = \sum_{d=1}^n f(d) \sum_{k=1}^{\lfloor n/d \rfloor} g(k)$$ 即： $$\sum_{i=1}^n h(i) = \sum_{d=1}^n f(d) S_g\left(\lfloor \frac{n}{d} \rfloor\right)$$

整理得杜教筛递推公式： $$g(1) S_f(n) = \sum_{i=1}^n h(i) - \sum_{d=2}^n f(d) S_g\left(\lfloor \frac{n}{d} \rfloor\right)$$

算法步骤：

  1. 输入：
    • 目标函数 $f(n)$（积性函数）
    • 辅助函数 $g(n)$，满足 $g(1) \neq 0$
    • 卷积函数 $h(n) = (f * g)(n)$，且 $h(n)$ 的前缀和容易计算
    • 正整数 $n$
  2. 预处理：
    1. 线性筛预处理 $f$ 的前 $m = \lfloor n^{2/3} \rfloor$ 个值，计算 $S_f(1)$ 到 $S_f(m)$
    2. 同样预处理 $g$ 和 $h$ 的前 $m$ 个值及其前缀和
  3. 递归计算： 对于给定的 $n$，计算 $S_f(n)$：
    1. 如果 $n \leq m$，直接返回预计算的 $S_f(n)$
    2. 如果 $n \gt m$，检查是否已计算过 $S_f(n)$（记忆化）
    3. 如果未计算，使用杜教筛公式： $$S_f(n) = \frac{1}{g(1)} \left( S_h(n) - \sum_{d=2}^n f(d) S_g\left(\lfloor \frac{n}{d} \rfloor\right) \right)$$
    4. 其中求和通过整数分块加速：$\lfloor n/d \rfloor$ 只有 $O(\sqrt{n})$ 个不同的值
    5. 递归计算需要的 $S_g(\lfloor n/d \rfloor)$ 值
  4. 优化技巧：
    • 使用哈希表记忆化已计算的 $S_f$ 值
    • 预处理前 $n^{2/3}$ 项以平衡时间和空间
    • 选择合适的辅助函数 $g$ 使得 $h$ 的前缀和容易计算
复杂度分析：
  • 预处理：$O(n^{2/3})$ 时间和空间
  • 递归计算：$O(n^{2/3})$ 时间（主定理分析）
  • 总复杂度：$O(n^{2/3})$，优于直接计算的 $O(n)$ 和朴素分块的 $O(\sqrt{n})$ 递归深度
  • 适用于 $n$ 高达 $10^{10}$ 甚至更大的情况

经典应用：

例1：计算欧拉函数 $\varphi(n)$ 的前缀和
设 $f(n) = \varphi(n)$，取 $g(n) = 1$（常值函数），则： $$h(n) = (\varphi * 1)(n) = \sum_{d|n} \varphi(d) = n$$ 因此 $S_h(n) = \sum_{i=1}^n i = \frac{n(n+1)}{2}$。
杜教筛公式为： $$S_\varphi(n) = \frac{n(n+1)}{2} - \sum_{d=2}^n S_\varphi\left(\lfloor \frac{n}{d} \rfloor\right)$$

例2：计算莫比乌斯函数 $\mu(n)$ 的前缀和
设 $f(n) = \mu(n)$，取 $g(n) = 1$，则： $$h(n) = (\mu * 1)(n) = \varepsilon(n) = \begin{cases} 1 & n=1 \\ 0 & n\gt1 \end{cases}$$ 因此 $S_h(n) = 1$。
杜教筛公式为： $$S_\mu(n) = 1 - \sum_{d=2}^n S_\mu\left(\lfloor \frac{n}{d} \rfloor\right)$$

例3：计算约数个数函数 $d(n)$ 的前缀和
设 $f(n) = d(n)$，取 $g(n) = \mu(n)$，则： $$h(n) = (d * \mu)(n) = 1$$ 因为 $d = 1 * 1$，所以 $d * \mu = 1 * 1 * \mu = 1 * \varepsilon = 1$。
因此 $S_h(n) = n$。
杜教筛公式为： $$S_d(n) = n - \sum_{d=2}^n \mu(d) S_d\left(\lfloor \frac{n}{d} \rfloor\right)$$ 但这里需要 $\mu$ 的前缀和，可以同时计算。

代码示例 (C++)


#include <iostream>
#include <vector>
#include <cmath>
#include <string>
#include <algorithm>
#include <unordered_map>
using namespace std;

using i128 = __int128;

// 打印 __int128（用于可能超过 64 位的前缀和）
void print_i128(i128 x) {
    if (x == 0) {
        cout << '0';
        return;
    }
    if (x < 0) {
        cout << '-';
        x = -x;
    }
    string s;
    while (x > 0) {
        s.push_back(char('0' + int(x % 10)));
        x /= 10;
    }
    reverse(s.begin(), s.end());
    cout << s;
}

// 线性筛预处理多个积性函数
struct Precomp {
    vector<int> primes;
    vector<int> mu, d;            // 莫比乌斯函数、约数个数
    vector<long long> phi;        // 欧拉函数
    vector<int> minPrime;         // 最小质因子
    vector<long long> sumMu, sumPhi, sumD;  // 前缀和
};

Precomp linearSieve(int limit) {
    Precomp pre;
    vector<bool> isPrime(limit + 1, true);
    pre.mu.assign(limit + 1, 0);
    pre.phi.assign(limit + 1, 0);
    pre.d.assign(limit + 1, 0);
    pre.minPrime.assign(limit + 1, 0);

    pre.mu[1] = 1;
    pre.phi[1] = 1;
    pre.d[1] = 1;

    for (int i = 2; i <= limit; i++) {
        if (isPrime[i]) {
            pre.primes.push_back(i);
            pre.mu[i] = -1;
            pre.phi[i] = i - 1;
            pre.d[i] = 2;
            pre.minPrime[i] = i;
        }
        for (int p : pre.primes) {
            long long num = 1LL * i * p;
            if (num > limit) break;
            isPrime[num] = false;
            pre.minPrime[num] = p;

            if (i % p == 0) {
                // i 包含质因子 p
                pre.mu[num] = 0;
                pre.phi[num] = pre.phi[i] * p;
                // 计算 d(num)：如果 i = p^a * m，则 num = p^(a+1) * m
                int temp = i, a = 0;
                while (temp % p == 0) {
                    temp /= p;
                    a++;
                }
                pre.d[num] = pre.d[i] / (a + 1) * (a + 2);
                break;
            } else {
                // i 和 p 互质
                pre.mu[num] = -pre.mu[i];
                pre.phi[num] = pre.phi[i] * (p - 1);
                pre.d[num] = pre.d[i] * 2;
            }
        }
    }

    pre.sumMu.assign(limit + 1, 0);
    pre.sumPhi.assign(limit + 1, 0);
    pre.sumD.assign(limit + 1, 0);
    for (int i = 1; i <= limit; i++) {
        pre.sumMu[i] = pre.sumMu[i - 1] + pre.mu[i];
        pre.sumPhi[i] = pre.sumPhi[i - 1] + pre.phi[i];
        pre.sumD[i] = pre.sumD[i - 1] + pre.d[i];
    }

    return pre;
}

// 杜教筛类
class DuSieve {
    long long n;
    int limit;
    Precomp pre;
    unordered_map<long long, long long> muMemo;
    unordered_map<long long, i128> phiMemo;
    unordered_map<long long, long long> dMemo;

public:
    DuSieve(long long n_) : n(n_) {
        limit = (int)pow((double)n, 2.0 / 3.0);  // 预处理大小
        pre = linearSieve(limit);
    }

    // 计算莫比乌斯函数前缀和 Sμ(x)
    long long sumMu(long long x) {
        if (x <= limit) return pre.sumMu[x];
        auto it = muMemo.find(x);
        if (it != muMemo.end()) return it->second;

        long long result = 1;  // S_h(n) = 1 (h = μ * 1 = ε)
        for (long long l = 2, r; l <= x; l = r + 1) {
            r = x / (x / l);
            result -= (r - l + 1) * sumMu(x / l);
        }
        muMemo[x] = result;
        return result;
    }

    // 计算欧拉函数前缀和 Sφ(x)
    i128 sumPhi(long long x) {
        if (x <= limit) return pre.sumPhi[x];
        auto it = phiMemo.find(x);
        if (it != phiMemo.end()) return it->second;

        // S_h(n) = n(n+1)/2 (h = φ * 1 = id)
        i128 result = (i128)x * (x + 1) / 2;
        for (long long l = 2, r; l <= x; l = r + 1) {
            r = x / (x / l);
            result -= (i128)(r - l + 1) * sumPhi(x / l);
        }
        phiMemo[x] = result;
        return result;
    }

    // 计算约数个数前缀和 Sd(x) - 需要同时计算 Sμ
    long long sumD(long long x) {
        if (x <= limit) return pre.sumD[x];
        auto it = dMemo.find(x);
        if (it != dMemo.end()) return it->second;

        long long result = x;  // h = d * μ = 1, 所以 S_h(n) = n
        for (long long l = 2, r; l <= x; l = r + 1) {
            r = x / (x / l);
            long long sumMuRange = sumMu(r) - sumMu(l - 1);
            result -= sumMuRange * sumD(x / l);
        }
        dMemo[x] = result;
        return result;
    }
};

int main() {
    cout << "=== 杜教筛示例 ===" << '\n';

    long long n = 1000000000LL;  // 10^9
    cout << "计算 n = " << n << " 的前缀和" << '\n';

    DuSieve du(n);
    cout << "Sμ(" << n << ") = " << du.sumMu(n) << '\n';
    cout << "Sφ(" << n << ") = ";
    print_i128(du.sumPhi(n));
    cout << '\n';
    cout << "Sd(" << n << ") = " << du.sumD(n) << '\n';

    // 验证小数值
    cout << '\n' << "=== 小数值验证 ===" << '\n';
    long long smallN = 1000;
    Precomp smallPre = linearSieve((int)smallN);
    DuSieve duSmall(smallN);
    long long sumMuSmall = duSmall.sumMu(smallN);
    i128 sumPhiSmall = duSmall.sumPhi(smallN);
    long long sumDSmall = duSmall.sumD(smallN);

    cout << "n=" << smallN << ":" << '\n';
    cout << "直接计算 Sμ = " << smallPre.sumMu[smallN] << ", 杜教筛 = " << sumMuSmall
         << ", 一致? " << (smallPre.sumMu[smallN] == sumMuSmall ? "true" : "false") << '\n';
    cout << "直接计算 Sφ = " << smallPre.sumPhi[smallN] << ", 杜教筛 = ";
    print_i128(sumPhiSmall);
    cout << ", 一致? " << ((i128)smallPre.sumPhi[smallN] == sumPhiSmall ? "true" : "false") << '\n';
    cout << "直接计算 Sd = " << smallPre.sumD[smallN] << ", 杜教筛 = " << sumDSmall
         << ", 一致? " << (smallPre.sumD[smallN] == sumDSmall ? "true" : "false") << '\n';

    // 多组规模对比
    cout << '\n' << "=== 多组规模对比 ===" << '\n';
    long long testNs[] = {1000000LL, 10000000LL, 100000000LL};
    for (long long testN : testNs) {
        DuSieve duTest(testN);
        cout << "Sφ(" << testN << ") ≈ ";
        print_i128(duTest.sumPhi(testN));
        cout << '\n';
    }

    // 计算更大的 n
    cout << '\n' << "=== 大数值计算 ===" << '\n';
    long long hugeN = 10000000000LL;  // 10^10
    cout << "尝试计算 n = " << hugeN << '\n';
    DuSieve duHuge(hugeN);
    cout << "Sφ(" << hugeN << ") ≈ ";
    print_i128(duHuge.sumPhi(hugeN));
    cout << '\n';

    return 0;
}

应用：

  1. 数论函数求和： 高效计算各种积性函数的前缀和，如 $\varphi(n)$、$\mu(n)$、$d(n)$、$\sigma(n)$ 等。
  2. 数论问题求解： 许多数论问题最终归结为计算某个数论函数的前缀和，杜教筛提供高效算法。
  3. 实际应用： 在要求计算大范围前缀和的题目中，杜教筛是常用工具。
  4. 数学研究： 在研究数论函数的渐近行为时，需要计算大量前缀和作为实验数据。
  5. 密码学： 在某些基于数论的密码算法分析中，需要计算相关函数的前缀和。

📝 优化与扩展：

1. 预处理大小选择： 通常取 $m = n^{2/3}$ 以达到理论最优复杂度 $O(n^{2/3})$。实际中可根据具体问题调整。
2. 记忆化优化： 使用哈希表存储已计算的结果，避免重复计算。
3. 多函数同时计算： 当需要计算多个相关函数的前缀和时，可以共享预处理和记忆化结果。
4. 递归深度控制： 通过预处理减少递归深度，提高实际运行效率。
5. 扩展性： 杜教筛可以推广到非积性函数，只要能找到合适的辅助函数 $g$ 使得 $h = f * g$ 的前缀和容易计算。
6. 局限性： 需要找到合适的辅助函数 $g$，且 $g$ 和 $h$ 的前缀和需要容易计算。对于某些函数可能不容易找到。

历史背景： 杜教筛由杜瑜皓在2016年提出，并在积性函数前缀和求解领域得到广泛应用。 它是对传统数论函数求和方法的重大改进，使得计算高达 $10^{10}$ 甚至更大的前缀和成为可能。
