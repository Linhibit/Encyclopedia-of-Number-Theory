算术级数中的素数定理 (Dirichlet's Theorem on Arithmetic Progressions)

定理内容：

设 $a$ 和 $d$ 为正整数，且 $\gcd(a, d) = 1$（即 $a$ 与 $d$ 互质）。 则算术级数（等差数列）： $$a, a + d, a + 2d, a + 3d, \dots$$ 中包含无穷多个素数。

等价表述：对于任意互质的正整数 $a$ 和 $d$，集合 $$\{a + nd \mid n \in \mathbb{N}\}$$ 中包含无穷多个素数。

更精确的渐近形式（狄利克雷密度定理）：在不超过 $x$ 的素数中，属于该算术级数的素数比例渐近于 $1/\varphi(d)$，其中 $\varphi$ 是欧拉函数。 即： $$\pi_{a,d}(x) \sim \frac{1}{\varphi(d)} \cdot \frac{x}{\ln x} \quad (x \to \infty)$$ 其中 $\pi_{a,d}(x)$ 表示不超过 $x$ 且满足 $p \equiv a \pmod{d}$ 的素数 $p$ 的个数。

证明思路：

狄利克雷的解析证明（概要）：

  1. 狄利克雷特征（Dirichlet characters）： 对于模 $d$，定义特征 $\chi: \mathbb{Z} \to \mathbb{C}$，满足：
    • $\chi(n+d) = \chi(n)$（周期性）
    • $\chi(mn) = \chi(m)\chi(n)$（完全积性）
    • $\chi(1) = 1$，且若 $\gcd(n, d) \gt 1$ 则 $\chi(n) = 0$
    共有 $\varphi(d)$ 个不同的特征。
  2. 狄利克雷L函数： 对于每个特征 $\chi$，定义： $$L(s, \chi) = \sum_{n=1}^{\infty} \frac{\chi(n)}{n^s}, \quad \Re(s) \gt 1$$
  3. 特征的正交关系： 利用特征的正交性，可以将算术级数中的素数计数表示为： $$\sum_{\substack{p \leq x \\ p \equiv a \ (\text{mod } d)}} 1 = \frac{1}{\varphi(d)} \sum_{\chi} \overline{\chi}(a) \sum_{p \leq x} \chi(p)$$
  4. 非平凡特征L函数的非零性： 证明对于非主特征 $\chi$，$L(1, \chi) \neq 0$。这是证明的关键步骤。
  5. 解析延拓与渐近分析： 通过分析L函数在 $s=1$ 附近的性质，得到素数分布的渐近公式。
  6. 无穷性结论： 从渐近公式直接推出算术级数中包含无穷多个素数。
证明难点：
  • 需要复杂的复分析和解析数论工具。
  • 关键步骤是证明 $L(1, \chi) \neq 0$，狄利克雷使用了巧妙的方法处理实数特征和复数特征。
  • 对于一般的 $d$，证明涉及群表示论和特征理论。

示例：

示例1： $a = 3$，$d = 4$（即形如 $4n+3$ 的素数）。

解法：

  1. $\gcd(3, 4) = 1$，满足定理条件。
  2. 该算术级数为：3, 7, 11, 15, 19, 23, 27, 31, 35, 39, 43, 47, ...
  3. 其中的素数：3, 7, 11, 19, 23, 31, 43, 47, 59, 67, 71, 79, 83, ...
  4. 定理断言存在无穷多个这样的素数。
  5. 事实上，形如 $4n+3$ 的素数有无穷多个，可以用初等方法证明（欧几里得变体）。

示例2： $a = 1$，$d = 4$（即形如 $4n+1$ 的素数）。

解法：

  1. $\gcd(1, 4) = 1$，满足定理条件。
  2. 该算术级数为：1, 5, 9, 13, 17, 21, 25, 29, 33, 37, 41, 45, 49, ...
  3. 其中的素数：5, 13, 17, 29, 37, 41, 53, 61, 73, 89, 97, 101, ...
  4. 形如 $4n+1$ 的素数也有无穷多个，但需要更复杂的方法证明。

示例3： $a = 6$，$d = 7$。

解法：

  1. $\gcd(6, 7) = 1$，满足定理条件。
  2. 该算术级数为：6, 13, 20, 27, 34, 41, 48, 55, 62, 69, 76, 83, 90, 97, ...
  3. 其中的素数：13, 41, 83, 97, 139, 167, 181, 223, 251, 293, 307, 349, ...
  4. 这些素数都满足 $p \equiv 6 \pmod{7}$。

反例： 如果 $\gcd(a, d) \gt 1$，则结论不成立。 例如 $a = 2$，$d = 4$，则所有项都是偶数，只有第一项 $2$ 是素数，之后没有素数。

代码示例 (C++)


#include <iostream>
#include <vector>
#include <cmath>
#include <iomanip>
#include <chrono>
#include <random>
using namespace std;

// 欧几里得算法求最大公约数
long long gcd(long long a, long long b) {
    while (b != 0) {
        long long t = a % b;
        a = b;
        b = t;
    }
    return a;
}

// 判断是否为素数（简单试除法）
bool isPrime(long long n) {
    if (n < 2) return false;
    if (n % 2 == 0) return n == 2;
    for (long long i = 3; i * i <= n; i += 2) {
        if (n % i == 0) return false;
    }
    return true;
}

// 生成算术级数中的素数
vector<long long> primesInArithmeticProgression(long long a, long long d, long long limit) {
    if (gcd(a, d) != 1) {
        cout << "警告：a=" << a << " 和 d=" << d << " 不互质，可能只有有限个素数。\n";
    }
    vector<long long> primes;
    long long n = 0;
    long long value = a;
    while (value <= limit) {
        if (isPrime(value)) primes.push_back(value);
        n++;
        value = a + n * d;
    }
    return primes;
}

// 计算不超过x的算术级数中的素数个数
long long countPrimesInProgression(long long a, long long d, long long x) {
    long long count = 0;
    long long n = 0;
    long long value = a;
    while (value <= x) {
        if (isPrime(value)) count++;
        n++;
        value = a + n * d;
    }
    return count;
}

// 欧拉函数 φ(n)
long long eulerPhi(long long n) {
    long long result = n;
    long long p = 2;
    while (p * p <= n) {
        if (n % p == 0) {
            while (n % p == 0) n /= p;
            result -= result / p;
        }
        p++;
    }
    if (n > 1) result -= result / n;
    return result;
}

// 快速幂取模（使用 __int128 防止乘法溢出）
long long modPow(long long base, long long exponent, long long modulus) {
    if (modulus == 1) return 0;
    long long result = 1;
    base %= modulus;
    while (exponent > 0) {
        if (exponent & 1) result = (__int128)result * base % modulus;
        exponent >>= 1;
        base = (__int128)base * base % modulus;
    }
    return result;
}

// 米勒-拉宾素性测试
bool isPrimeMillerRabin(long long n, int k = 5) {
    if (n < 2) return false;
    if (n == 2 || n == 3) return true;
    if (n % 2 == 0) return false;

    // 将 n-1 分解为 2^s * d
    long long s = 0;
    long long d = n - 1;
    while (d % 2 == 0) {
        d /= 2;
        s++;
    }

    static mt19937 rng(123456789); // 固定种子，结果可复现
    for (int i = 0; i < k; i++) {
        long long a = 2 + (long long)(rng() % (n - 4));
        long long x = modPow(a, d, n);

        if (x == 1 || x == n - 1) continue;

        bool ok = false;
        for (long long r = 1; r < s; r++) {
            x = modPow(x, 2, n);
            if (x == n - 1) {
                ok = true;
                break;
            }
        }

        if (ok) continue;
        return false;
    }

    return true;
}

int main() {
    cout << "=== 算术级数中的素数定理示例 ===\n";

    // 示例1：形如 4n+3 的素数
    cout << "\n1. 形如 4n+3 的素数（不超过100）：\n";
    vector<long long> primes1 = primesInArithmeticProgression(3, 4, 100);
    cout << "素数列表：";
    for (size_t i = 0; i < primes1.size(); i++) {
        if (i) cout << ", ";
        cout << primes1[i];
    }
    cout << "\n个数：" << primes1.size() << "\n";

    // 示例2：形如 4n+1 的素数
    cout << "\n2. 形如 4n+1 的素数（不超过100）：\n";
    vector<long long> primes2 = primesInArithmeticProgression(1, 4, 100);
    cout << "素数列表：";
    for (size_t i = 0; i < primes2.size(); i++) {
        if (i) cout << ", ";
        cout << primes2[i];
    }
    cout << "\n个数：" << primes2.size() << "\n";

    // 示例3：验证密度定理
    cout << "\n3. 验证狄利克雷密度定理（a=3, d=4）：\n";
    for (long long x : {100LL, 1000LL, 10000LL, 100000LL}) {
        long long actual = countPrimesInProgression(3, 4, x);
        double expected = (1.0 / eulerPhi(4)) * (x / log(x));
        cout << "x=" << x << ": 实际素数数=" << actual
             << ", 预期≈" << fixed << setprecision(2) << expected
             << ", 比例=" << fixed << setprecision(3) << actual / expected << "\n";
    }

    // 比较不同算术级数的素数分布
    cout << "\n4. 不同算术级数的素数分布比较（不超过10000）：\n";
    for (long long d : {3LL, 4LL, 5LL, 6LL, 7LL, 8LL}) {
        long long phi = eulerPhi(d);
        cout << "\n模 d=" << d << " (φ(" << d << ")=" << phi << "):\n";
        long long total = countPrimesInProgression(1, 1, 10000);
        for (long long a = 1; a < d; a++) {
            if (gcd(a, d) == 1) {
                long long count = countPrimesInProgression(a, d, 10000);
                double proportion = (double)count / total;
                double expectedProp = 1.0 / phi;
                cout << "  a=" << a << ": " << count << "个素数，占比="
                     << fixed << setprecision(4) << proportion
                     << "，期望占比=" << expectedProp << "\n";
            }
        }
    }

    // 寻找算术级数中的大素数
    cout << "\n5. 寻找算术级数中的大素数（a=1, d=1000）：\n";
    long long largePrime = 0;
    long long n = 0;
    while (largePrime == 0 && n < 1000) {
        long long value = 1 + n * 1000;
        if (isPrime(value)) {
            largePrime = value;
            cout << "找到素数：" << largePrime << " = 1 + " << n << " × 1000\n";
        }
        n++;
    }

    // 性能测试：计算大范围内的素数
    cout << "\n6. 性能测试：计算不超过1000000的形如 4n+3 的素数\n";
    auto start = chrono::steady_clock::now();
    long long count = countPrimesInProgression(3, 4, 1000000);
    auto end = chrono::steady_clock::now();
    double ms = chrono::duration<double, milli>(end - start).count();
    cout << "计算时间：" << fixed << setprecision(2) << ms << " 毫秒\n";
    cout << "素数个数：" << count << "\n";

    // 使用米勒-拉宾测试验证大数
    cout << "\n7. 使用米勒-拉宾测试验证大数\n";
    for (long long num : {1000003LL, 1000033LL, 1000037LL, 1000039LL}) {
        bool isPrimeMR = isPrimeMillerRabin(num, 10);
        cout << num << " 是素数？ " << (isPrimeMR ? "true" : "false") << "\n";
    }

    return 0;
}

应用：

  1. 数论研究： 定理是解析数论的核心成果之一，推动了L函数和模形式理论的发展。
  2. 素数生成： 在密码学中，需要生成大素数，算术级数提供了寻找素数的有效途径。
  3. 算法设计： 基于算术级数的素数筛选算法，如狄利克雷特征用于加速计算。
  4. 数学教育： 展示了解析方法在数论中的强大力量，是数论课程的重要教学内容。
  5. 理论计算机科学： 在复杂性理论和算法分析中，算术级数中的素数分布有重要应用。

📝 历史背景：

狄利克雷定理由德国数学家约翰·彼得·古斯塔夫·勒热纳·狄利克雷于1837年证明。 这是解析数论早期的重要成就，首次成功地将解析方法应用于数论问题。

狄利克雷的证明引入了狄利克雷特征和L函数，这些工具后来成为现代数论的核心。 定理推广了欧几里得关于素数无穷多的结果，显示了素数在算术级数中的均匀分布。

注意： 1. 定理要求 $\gcd(a, d) = 1$，否则算术级数中最多只有一个素数。
2. 定理是"存在无穷多个"的定性结果，密度定理给出了定量的渐近分布。
3. 对于某些特殊的 $d$（如 $d=4$），可以用初等方法证明，但一般情况需要解析方法。
4. 定理在广义黎曼猜想下有更精确的误差项估计。
