狄利克雷卷积 (Dirichlet Convolution)

定义：

设 $f$ 和 $g$ 是两个数论函数（定义在正整数集上的函数）。 它们的狄利克雷卷积 $f * g$ 定义为： $$(f * g)(n) = \sum_{d \mid n} f(d) g\left(\frac{n}{d}\right)$$ 其中求和遍历 $n$ 的所有正因子 $d$。

等价形式： $$(f * g)(n) = \sum_{ab = n} f(a) g(b)$$ 其中 $a, b$ 取遍满足 $ab = n$ 的正整数对。

代数性质：

  1. 交换律： $f * g = g * f$
  2. 结合律： $(f * g) * h = f * (g * h)$
  3. 分配律： $f * (g + h) = f * g + f * h$
  4. 单位元： 定义数论函数 $\varepsilon$： $$\varepsilon(n) = \begin{cases} 1 & \text{如果 } n = 1 \\ 0 & \text{如果 } n \gt 1 \end{cases}$$ 则对于任意数论函数 $f$，有 $f * \varepsilon = \varepsilon * f = f$。
  5. 逆元： 如果 $f(1) \neq 0$，则存在唯一的数论函数 $f^{-1}$（称为狄利克雷逆）使得： $$f * f^{-1} = f^{-1} * f = \varepsilon$$

这些性质表明，所有数论函数在狄利克雷卷积下构成一个交换环， 其中加法是函数的逐点加法，乘法是狄利克雷卷积。

常见数论函数的卷积：

函数 符号 定义 重要卷积关系
常数函数 $1$ $1(n) = 1$ $(1 * \mu) = \varepsilon$
恒等函数 $\text{id}$ $\text{id}(n) = n$ $\text{id} = \varphi * 1$
欧拉函数 $\varphi$ 欧拉函数 $\varphi = \text{id} * \mu$
约数个数 $d$ 或 $\tau$ $d(n) = \sum_{d|n} 1$ $d = 1 * 1$
约数和 $\sigma$ $\sigma(n) = \sum_{d|n} d$ $\sigma = \text{id} * 1$
莫比乌斯函数 $\mu$ 莫比乌斯函数 $\mu = 1^{-1}$
单位函数 $\varepsilon$ $\varepsilon(1)=1$, $\varepsilon(n)=0$ ($n\gt1$) $f * \varepsilon = f$

莫比乌斯反演的卷积形式：

莫比乌斯反演公式可以用狄利克雷卷积简洁表达：

定理： 设 $f$ 和 $g$ 是数论函数，则： $$f = g * 1 \quad \text{当且仅当} \quad g = f * \mu$$

证明： 因为 $\mu$ 是 $1$ 的狄利克雷逆，即 $1 * \mu = \mu * 1 = \varepsilon$。 如果 $f = g * 1$，则两边同时卷积 $\mu$： $$f * \mu = (g * 1) * \mu = g * (1 * \mu) = g * \varepsilon = g$$ 反之亦然。

这个形式比原始的求和形式更加简洁优美，并且容易推广。

示例：

示例1： 验证 $\varphi = \text{id} * \mu$。

计算： 对于 $n=6$：

  • $(\text{id} * \mu)(6) = \sum_{d|6} \text{id}(d) \mu(6/d)$
  • $6$ 的因子：$1, 2, 3, 6$
  • 计算：
    • $d=1$: $\text{id}(1) \mu(6) = 1 \times \mu(6) = 1 \times 1 = 1$（因为 $6=2\times3$，$\mu(6)=1$）
    • $d=2$: $\text{id}(2) \mu(3) = 2 \times (-1) = -2$
    • $d=3$: $\text{id}(3) \mu(2) = 3 \times (-1) = -3$
    • $d=6$: $\text{id}(6) \mu(1) = 6 \times 1 = 6$
  • 总和：$1 + (-2) + (-3) + 6 = 2$
  • $\varphi(6) = 2$，确实相等。

示例2： 计算 $(1 * \varphi)(8)$。

计算： $$(1 * \varphi)(8) = \sum_{d|8} \varphi(d)$$ $8$ 的因子：$1, 2, 4, 8$

  • $\varphi(1) = 1$
  • $\varphi(2) = 1$
  • $\varphi(4) = 2$
  • $\varphi(8) = 4$
总和：$1+1+2+4=8$。 注意：$(1 * \varphi)(n) = n$ 是一个恒等式，因为 $\varphi = \text{id} * \mu$， 所以 $1 * \varphi = 1 * (\text{id} * \mu) = (1 * \mu) * \text{id} = \varepsilon * \text{id} = \text{id}$。

代码示例 (C++)


#include <iostream>
#include <vector>
#include <algorithm>
#include <functional>
using namespace std;

// 获取 n 的所有因子（升序）
vector<long long> getDivisors(long long n) {
    vector<long long> divisors;
    for (long long i = 1; i * i <= n; i++) {
        if (n % i == 0) {
            divisors.push_back(i);
            if (i != n / i) divisors.push_back(n / i);
        }
    }
    sort(divisors.begin(), divisors.end());
    return divisors;
}

// 狄利克雷卷积实现
long long dirichletConvolution(function<long long(long long)> f,
                               function<long long(long long)> g,
                               long long n) {
    vector<long long> divisors = getDivisors(n);
    long long result = 0;
    for (long long d : divisors) {
        result += f(d) * g(n / d);
    }
    return result;
}

// 常见数论函数的实现

// 常数函数 1
long long constantOne(long long) {
    return 1;
}

// 恒等函数 id(n) = n
long long identity(long long n) {
    return n;
}

// 欧拉函数 φ(n)
long long eulerPhi(long long n) {
    long long result = n;
    long long temp = n;
    for (long long p = 2; p * p <= temp; p++) {
        if (temp % p == 0) {
            while (temp % p == 0) temp /= p;
            result -= result / p;
        }
    }
    if (temp > 1) {
        result -= result / temp;
    }
    return result;
}

// 约数个数 d(n)
long long divisorCount(long long n) {
    long long count = 0;
    for (long long i = 1; i * i <= n; i++) {
        if (n % i == 0) {
            count++;
            if (i != n / i) count++;
        }
    }
    return count;
}

// 约数和 σ(n)
long long divisorSum(long long n) {
    long long sum = 0;
    for (long long i = 1; i * i <= n; i++) {
        if (n % i == 0) {
            sum += i;
            if (i != n / i) sum += n / i;
        }
    }
    return sum;
}

// 莫比乌斯函数 μ(n)
long long mobius(long long n) {
    if (n <= 0) return 0;
    if (n == 1) return 1;
    long long primeCount = 0;
    long long temp = n;
    for (long long p = 2; p * p <= temp; p++) {
        if (temp % p == 0) {
            primeCount++;
            temp /= p;
            if (temp % p == 0) {
                return 0; // 有平方因子
            }
        }
    }
    if (temp > 1) {
        primeCount++;
    }
    return primeCount % 2 == 0 ? 1 : -1;
}

// 单位函数 ε(n)
long long unitEpsilon(long long n) {
    return n == 1 ? 1 : 0;
}

// 狄利克雷逆：递归计算 f^{-1}(n)
// 公式：f^{-1}(1) = 1/f(1); 对于 n>1: f^{-1}(n) = -1/f(1) * Σ_{d|n, d<n} f(n/d) * f^{-1}(d)
vector<long long> dirichletInverse(function<long long(long long)> f, long long n) {
    vector<long long> inv(n + 1, 0);
    if (f(1) == 0) {
        cerr << "f(1)=0，逆不存在" << '\n';
        return inv;
    }
    inv[1] = 1 / f(1);
    for (long long i = 2; i <= n; i++) {
        long long sum = 0;
        vector<long long> divisors = getDivisors(i);
        for (long long d : divisors) {
            if (d < i) sum += f(i / d) * inv[d];
        }
        inv[i] = -sum / f(1);
    }
    return inv;
}

int main() {
    cout << "=== 狄利克雷卷积示例 ===" << '\n';

    // 示例1：验证 φ = id * μ
    cout << "\n1. 验证 φ = id * μ (n=1..10):" << '\n';
    for (long long n = 1; n <= 10; n++) {
        long long phiDirect = eulerPhi(n);
        long long phiConv = dirichletConvolution(identity, mobius, n);
        cout << "n=" << n << ": φ(" << n << ")=" << phiDirect
             << ", (id*μ)(" << n << ")=" << phiConv
             << ", " << (phiDirect == phiConv ? "✓" : "✗") << '\n';
    }

    // 示例2：验证 1 * φ = id
    cout << "\n2. 验证 1 * φ = id (n=1..10):" << '\n';
    for (long long n = 1; n <= 10; n++) {
        long long idDirect = n;
        long long conv = dirichletConvolution(constantOne, eulerPhi, n);
        cout << "n=" << n << ": id(" << n << ")=" << idDirect
             << ", (1*φ)(" << n << ")=" << conv
             << ", " << (idDirect == conv ? "✓" : "✗") << '\n';
    }

    // 示例3：验证 d = 1 * 1
    cout << "\n3. 验证 d = 1 * 1 (n=1..10):" << '\n';
    for (long long n = 1; n <= 10; n++) {
        long long dDirect = divisorCount(n);
        long long conv = dirichletConvolution(constantOne, constantOne, n);
        cout << "n=" << n << ": d(" << n << ")=" << dDirect
             << ", (1*1)(" << n << ")=" << conv
             << ", " << (dDirect == conv ? "✓" : "✗") << '\n';
    }

    // 示例4：验证 σ = id * 1
    cout << "\n4. 验证 σ = id * 1 (n=1..10):" << '\n';
    for (long long n = 1; n <= 10; n++) {
        long long sigmaDirect = divisorSum(n);
        long long conv = dirichletConvolution(identity, constantOne, n);
        cout << "n=" << n << ": σ(" << n << ")=" << sigmaDirect
             << ", (id*1)(" << n << ")=" << conv
             << ", " << (sigmaDirect == conv ? "✓" : "✗") << '\n';
    }

    // 示例5：计算常数函数1的狄利克雷逆（应该是μ）
    cout << "\n5. 计算常数函数1的狄利克雷逆（应该是μ）:" << '\n';
    vector<long long> inv1 = dirichletInverse(constantOne, 20);
    cout << "n\t计算逆\tμ(n)" << '\n';
    for (long long i = 1; i <= 20; i++) {
        cout << i << '\t' << inv1[i] << '\t' << mobius(i) << '\n';
    }

    // 示例6：验证常数函数1与其逆的卷积是ε
    cout << "\n6. 验证常数函数1与其逆的卷积是ε:" << '\n';
    for (long long n = 1; n <= 10; n++) {
        long long conv = dirichletConvolution(constantOne,
            [&inv1](long long x) { return inv1[x]; }, n);
        long long expected = unitEpsilon(n);
        cout << "n=" << n << ": (1*1^{-1})(" << n << ")=" << conv
             << ", ε(" << n << ")=" << expected
             << ", " << (conv == expected ? "✓" : "✗") << '\n';
    }

    return 0;
}

应用：

  1. 数论函数研究： 狄利克雷卷积为数论函数提供了代数结构，便于研究函数之间的关系。
  2. 莫比乌斯反演： 用卷积形式表达更加简洁，便于推广和应用。
  3. 解析数论： 狄利克雷级数的乘法对应狄利克雷卷积。
  4. 组合数学： 某些计数问题可以转化为数论函数和狄利克雷卷积。
  5. 算法设计： 利用卷积性质可以设计更高效的算法计算数论函数。

📝 狄利克雷级数：

狄利克雷卷积与狄利克雷级数有密切关系。数论函数 $f$ 的狄利克雷级数定义为： $$F(s) = \sum_{n=1}^\infty \frac{f(n)}{n^s}$$ 其中 $s$ 是复数。

两个数论函数的狄利克雷卷积对应于它们狄利克雷级数的普通乘积： $$(f * g)(n) \quad \text{对应} \quad F(s) \cdot G(s)$$ 这使得我们可以用解析方法研究数论函数。

例如，黎曼zeta函数 $\zeta(s) = \sum_{n=1}^\infty \frac{1}{n^s}$ 对应常数函数 $1$， 而 $\frac{1}{\zeta(s)} = \sum_{n=1}^\infty \frac{\mu(n)}{n^s}$ 对应莫比乌斯函数 $\mu$。
