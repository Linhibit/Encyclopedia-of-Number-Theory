费马二平方和定理 (Fermat's Two-Square Theorem)

定理内容：

一个奇素数 $p$ 可以表示为两个整数的平方和，即： $$p = a^2 + b^2 \quad (a, b \in \mathbb{Z})$$ 当且仅当 $p \equiv 1 \pmod{4}$。

等价表述：

  1. 奇素数 $p$ 可以表示为两个平方数之和当且仅当 $p = 2$ 或 $p \equiv 1 \pmod{4}$。
  2. 奇素数 $p$ 在整数环 $\mathbb{Z}[i]$（高斯整数环）中不是素元当且仅当 $p \equiv 1 \pmod{4}$。

例子：

  • $5 = 1^2 + 2^2$，且 $5 \equiv 1 \pmod{4}$ ✓
  • $13 = 2^2 + 3^2$，且 $13 \equiv 1 \pmod{4}$ ✓
  • $17 = 1^2 + 4^2$，且 $17 \equiv 1 \pmod{4}$ ✓
  • $3 = ?$，但 $3 \equiv 3 \pmod{4}$，不能表示为两个平方和 ✓
  • $7 = ?$，但 $7 \equiv 3 \pmod{4}$，不能表示为两个平方和 ✓

证明思路：

必要性证明（较简单）：

  1. 假设 $p = a^2 + b^2$，其中 $a, b$ 为整数。
  2. 平方数模4只能是0或1，所以 $a^2 + b^2$ 模4只能是0, 1, 2。
  3. 由于 $p$ 是奇素数，不能是偶数，所以 $a^2 + b^2 \equiv 1 \pmod{4}$。
  4. 因此 $p \equiv 1 \pmod{4}$ 是必要条件。

充分性证明（需要更多工具）：

  1. 若 $p \equiv 1 \pmod{4}$，则存在整数 $x$ 使得 $x^2 \equiv -1 \pmod{p}$。
  2. 考虑所有形如 $a + bx$ 的数，其中 $0 \le a, b \le \lfloor\sqrt{p}\rfloor$。
  3. 根据抽屉原理，存在两对不同的 $(a,b)$ 使得 $a + bx \equiv a' + b'x \pmod{p}$。
  4. 设 $u = a - a'$，$v = b - b'$，则 $u + vx \equiv 0 \pmod{p}$。
  5. 因此 $p \mid (u^2 + v^2)$，且 $0 \lt u^2 + v^2 \lt 2p$。
  6. 所以 $u^2 + v^2 = p$，即 $p$ 可以表示为两个平方和。

另一种证明（使用高斯整数）：

  1. 在 $\mathbb{Z}[i]$ 中，$p \mid (x^2 + 1) = (x + i)(x - i)$。
  2. 由于 $p \nmid (x \pm i)$，$p$ 不是 $\mathbb{Z}[i]$ 中的素元。
  3. 因此 $p = \alpha \beta$，其中 $\alpha, \beta \in \mathbb{Z}[i]$ 且不是单位。
  4. 取范数：$p^2 = N(p) = N(\alpha)N(\beta)$，所以 $N(\alpha) = N(\beta) = p$。
  5. 设 $\alpha = a + bi$，则 $N(\alpha) = a^2 + b^2 = p$。

示例与应用：

示例1： 找出 $p = 29$ 的二平方和表示。

解法：

  1. 首先验证 $29 \equiv 1 \pmod{4}$，所以存在表示。
  2. 尝试 $a = 1$：$29 - 1^2 = 28$，不是完全平方数。
  3. $a = 2$：$29 - 4 = 25$，是完全平方数 $5^2$。
  4. 所以 $29 = 2^2 + 5^2$。
  5. 另一种表示：$29 = 5^2 + 2^2$（相同）。

示例2： 判断 $p = 97$ 是否可以表示为二平方和。

解法：

  1. $97 \equiv 1 \pmod{4}$，所以可以表示。
  2. 寻找表示：$97 - 1^2 = 96$（不是平方），$97 - 4 = 93$（不是），$97 - 9 = 88$（不是）
  3. $97 - 16 = 81 = 9^2$，所以 $97 = 4^2 + 9^2$。

示例3： 合数的二平方和表示。

注意： 费马二平方和定理只适用于素数。对于合数：

  • $10 = 1^2 + 3^2$，但 $10 \equiv 2 \pmod{4}$
  • $25 = 0^2 + 5^2 = 3^2 + 4^2$，且 $25 \equiv 1 \pmod{4}$
  • 合数的二平方和表示有更复杂的规律，涉及素因子分解。

布拉马古普塔-斐波那契恒等式： $$(a^2 + b^2)(c^2 + d^2) = (ac + bd)^2 + (ad - bc)^2 = (ac - bd)^2 + (ad + bc)^2$$ 这个恒等式说明，两个可以表示为二平方和的数的乘积也可以表示为二平方和（且有两种方式）。

代码示例 (C++)


#include <iostream>
#include <vector>
#include <string>
#include <cmath>
#include <cstdlib>
using namespace std;
using ll = long long;

// 二平方和表示的结构体
struct Repr {
    ll a;
    ll b;
};

// 判断一个数是否为素数
bool isPrime(ll n) {
    if (n <= 1) return false;
    if (n <= 3) return true;
    if (n % 2 == 0 || n % 3 == 0) return false;
    for (ll i = 5; i * i <= n; i += 6) {
        if (n % i == 0 || n % (i + 2) == 0) return false;
    }
    return true;
}

// 使用费马二平方和定理判断素数是否可以表示为二平方和
bool canBeRepresentedAsTwoSquares(ll p) {
    if (!isPrime(p)) {
        cout << "注意：" << p << " 不是素数，费马二平方和定理只适用于素数\n";
    }
    
    // 对于 p = 2，有 2 = 1^2 + 1^2
    if (p == 2) return true;
    
    // 奇素数可以表示为二平方和当且仅当 p ≡ 1 mod 4
    return (p % 4 == 1);
}

// 寻找二平方和表示（适用于满足条件的素数）
Repr findTwoSquaresRepresentation(ll p) {
    Repr r = {-1, -1};
    if (p == 2) { r.a = 1; r.b = 1; return r; }
    
    // 检查条件
    if (p % 4 != 1) {
        return r; // 不能表示为二平方和
    }
    
    // 寻找 a 使得 p - a^2 是完全平方数
    for (ll a = 1; a * a <= p; a++) {
        ll remainder = p - a * a;
        ll b = llround(sqrt((double)remainder));
        
        // 检查是否为整数平方
        if (b * b == remainder) {
            r.a = a;
            r.b = b;
            return r;
        }
    }
    
    return r; // 理论上不会执行到这里
}

// 使用更高效的算法（基于欧几里得算法）
Repr findTwoSquaresEfficient(ll p) {
    Repr r = {-1, -1};
    if (p == 2) { r.a = 1; r.b = 1; return r; }
    if (p % 4 != 1) return r;
    
    // 步骤1：寻找 x 使得 x^2 ≡ -1 mod p
    ll x = -1;
    for (ll i = 2; i < p; i++) {
        if ((i * i) % p == p - 1) {
            x = i;
            break;
        }
    }
    
    // 步骤2：使用欧几里得算法
    ll a = p;
    ll b = x;
    while (b * b > p) {
        ll rem = a % b;
        a = b;
        b = rem;
    }
    
    ll b2 = llround(sqrt((double)(p - b * b)));
    if (b2 * b2 == p - b * b) {
        r.a = b;
        r.b = b2;
        return r;
    }
    
    return r;
}

// 布拉马古普塔-斐波那契恒等式
void brahmaguptaFibonacci(ll a, ll b, ll c, ll d) {
    ll product = (a*a + b*b) * (c*c + d*d);
    ll ac_bd1 = a*c + b*d;
    ll ad_bc1 = abs(a*d - b*c);
    ll ac_bd2 = abs(a*c - b*d);
    ll ad_bc2 = a*d + b*c;
    
    cout << "(" << a << "² + " << b << "²) × (" << c << "² + " << d << "²) = " << product << '\n';
    cout << "表示1: (" << a << "×" << c << " + " << b << "×" << d << ")² + (" << a << "×" << d << " - " << b << "×" << c << ")² = " << ac_bd1 << "² + " << ad_bc1 << "²\n";
    cout << "表示2: (" << a << "×" << c << " - " << b << "×" << d << ")² + (" << a << "×" << d << " + " << b << "×" << c << ")² = " << ac_bd2 << "² + " << ad_bc2 << "²\n";
    cout << "验证: " << (ac_bd1*ac_bd1 + ad_bc1*ad_bc1) << " = " << product << '\n';
    cout << "验证: " << (ac_bd2*ac_bd2 + ad_bc2*ad_bc2) << " = " << product << '\n';
    cout << string(40, '-') << '\n';
}

// 合数的二平方和表示
vector<Repr> findTwoSquaresForComposite(ll n) {
    vector<Repr> representations;
    
    // 简单搜索（适用于较小的数）
    ll limit = (ll)sqrt((double)n);
    for (ll a = 0; a <= limit; a++) {
        ll a2 = a * a;
        ll remainder = n - a2;
        if (remainder < 0) break;
        
        ll b = llround(sqrt((double)remainder));
        if (b * b == remainder) {
            Repr r = {a, b};
            representations.push_back(r);
        }
    }
    
    return representations;
}

// 判断素数在 Z[i] 中是否可约
bool isPrimeInGaussianIntegers(ll p) {
    if (p == 2) return false; // 2 = (1+i)(1-i)，可约
    return (p % 4 == 1); // p ≡ 1 mod 4 时可约
}

int main() {
    // 示例使用
    cout << "=== 费马二平方和定理示例 ===\n";
    
    // 测试一些素数
    vector<ll> testPrimes = {2, 3, 5, 7, 13, 17, 29, 37, 41, 97, 101, 109, 113};
    
    cout << "素数\t≡1 mod 4?\t可表示为二平方和?\t表示\n";
    cout << string(60, '-') << '\n';
    
    for (ll p : testPrimes) {
        bool canBe = canBeRepresentedAsTwoSquares(p);
        Repr representation = {-1, -1};
        
        if (canBe) {
            representation = findTwoSquaresEfficient(p);
        }
        
        cout << p << '\t' << (p % 4 == 1 ? "是" : "否") << "\t\t"
             << (canBe ? "是" : "否") << "\t\t\t";
        if (representation.a >= 0) {
            cout << representation.a << "² + " << representation.b << "²";
        } else {
            cout << "无";
        }
        cout << '\n';
    }
    
    // 验证布拉马古普塔-斐波那契恒等式
    cout << "\n=== 布拉马古普塔-斐波那契恒等式验证 ===\n";
    
    // 测试几个例子
    brahmaguptaFibonacci(1, 2, 2, 3);  // (1²+2²)×(2²+3²) = 5×13 = 65
    brahmaguptaFibonacci(2, 3, 1, 4);  // (2²+3²)×(1²+4²) = 13×17 = 221
    brahmaguptaFibonacci(3, 4, 2, 5);  // (3²+4²)×(2²+5²) = 25×29 = 725
    
    // 合数的二平方和表示
    cout << "\n=== 合数的二平方和表示 ===\n";
    
    vector<ll> compositeNumbers = {10, 25, 50, 65, 85, 100};
    for (ll n : compositeNumbers) {
        vector<Repr> reps = findTwoSquaresForComposite(n);
        cout << n << " 的二平方和表示:\n";
        if (!reps.empty()) {
            for (const Repr& rep : reps) {
                cout << "  " << rep.a << "² + " << rep.b << "²\n";
            }
        } else {
            cout << "  无（注意：有些数如6, 14等不能表示为二平方和）\n";
        }
    }
    
    // 高斯整数环上的应用
    cout << "\n=== 高斯整数环上的应用 ===\n";
    
    vector<ll> gaussianPrimes = {2, 3, 5, 7, 11, 13, 17, 19};
    for (ll p : gaussianPrimes) {
        bool reducible = isPrimeInGaussianIntegers(p);
        cout << p << " 在 Z[i] 中" << (reducible ? "可约" : "不可约（是素元）") << '\n';
    }
    
    return 0;
}

应用与推广：

  1. 高斯整数环： 费马二平方和定理等价于在 $\mathbb{Z}[i]$ 中，奇素数 $p$ 是素元当且仅当 $p \equiv 3 \pmod{4}$。当 $p \equiv 1 \pmod{4}$ 时，$p$ 可以分解为两个高斯整数的乘积。
  2. 密码学： 在某些基于格的密码系统中，二平方和表示可以用于参数选择。
  3. 数论研究： 费马二平方和定理是研究平方和表示问题的起点，后来发展出更一般的定理（如四平方和定理）。
  4. 组合数学： 与分割和表示问题相关。
  5. 几何： 在圆上的整点问题中有应用。

📝 历史背景与扩展：

费马二平方和定理由皮埃尔·德·费马在1640年提出，但费马没有给出证明。欧拉在1747年给出了第一个完整的证明。

扩展： 1. 对于合数 $n$，可以表示为二平方和的充要条件是：$n$ 的每个形如 $4k+3$ 的素因子在 $n$ 的质因数分解中出现偶数次。
2. 若 $n$ 可以表示为二平方和，则 $n$ 的表示方法数 $r_2(n)$ 由雅可比公式给出： $$r_2(n) = 4(d_1(n) - d_3(n))$$ 其中 $d_1(n)$ 是 $n$ 的 $4k+1$ 型因子的个数，$d_3(n)$ 是 $n$ 的 $4k+3$ 型因子的个数。
3. 费马二平方和定理可以推广到其他二次型，如 $x^2 + 2y^2$、$x^2 + 3y^2$ 等。

注意： 费马二平方和定理是狄利克雷定理的一个特例，后者描述素数在算术级数中的分布。
