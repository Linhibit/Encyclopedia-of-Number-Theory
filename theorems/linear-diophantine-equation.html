线性丢番图方程 (Linear Diophantine Equation)

方程形式：

二元一次线性丢番图方程的标准形式为： $$ax + by = c$$ 其中 $a, b, c$ 是给定的整数，$x$ 和 $y$ 是未知整数。

解的存在性定理： 方程 $ax + by = c$ 有整数解当且仅当 $\gcd(a, b) \mid c$。

通解公式： 若 $(x_0, y_0)$ 是一个特解，则所有整数解为： $$ \begin{cases} x = x_0 + \dfrac{b}{d} t \\[8pt] y = y_0 - \dfrac{a}{d} t \end{cases} $$ 其中 $d = \gcd(a, b)$，$t$ 为任意整数。

理论基础：

  1. 贝祖等式： 对于任意整数 $a, b$，存在整数 $x, y$ 使得 $ax + by = \gcd(a, b)$。 这是线性丢番图方程解存在性的基础。
  2. 必要性证明： 如果 $ax + by = c$ 有整数解，设 $d = \gcd(a, b)$，则 $d \mid a$ 且 $d \mid b$， 因此 $d \mid (ax + by) = c$。
  3. 充分性证明： 如果 $d \mid c$，设 $c = kd$。由贝祖等式，存在 $x_1, y_1$ 使得 $ax_1 + by_1 = d$。 两边乘以 $k$ 得 $a(kx_1) + b(ky_1) = kd = c$，因此 $(kx_1, ky_1)$ 是一个特解。
  4. 通解推导： 设 $(x_0, y_0)$ 和 $(x, y)$ 都是解，则 $a(x - x_0) + b(y - y_0) = 0$， 所以 $a(x - x_0) = -b(y - y_0)$。令 $d = \gcd(a, b)$，$a = da'$, $b = db'$， 其中 $\gcd(a', b') = 1$。于是 $a'(x - x_0) = -b'(y - y_0)$， 由于 $\gcd(a', b') = 1$，有 $a' \mid (y - y_0)$，设 $y - y_0 = a't$， 则 $x - x_0 = -b't$，整理即得通解公式。

求解方法：扩展欧几里得算法

步骤：

  1. 检查解的存在性： 计算 $d = \gcd(a, b)$，如果 $d \nmid c$，则无整数解。
  2. 约化方程： 如果 $d \mid c$，令 $a' = a/d$, $b' = b/d$, $c' = c/d$， 则原方程等价于 $a'x + b'y = c'$，且 $\gcd(a', b') = 1$。
  3. 求特解： 使用扩展欧几里得算法求 $a'x_1 + b'y_1 = 1$ 的解 $(x_1, y_1)$， 则 $(c'x_1, c'y_1)$ 是 $a'x + b'y = c'$ 的一个特解。
  4. 写出通解： 原方程的通解为： $$ \begin{cases} x = c'x_1 + b't \\ y = c'y_1 - a't \end{cases} \quad t \in \mathbb{Z} $$
  5. 非负解/正整数解： 如果需要非负解或正整数解，可对 $t$ 的取值范围加以限制。

示例：

示例1： 求解 $21x + 14y = 35$。

解法：

  1. 计算 $\gcd(21, 14) = 7$，$7 \mid 35$，所以有解。
  2. 约化：$21/7 = 3$，$14/7 = 2$，$35/7 = 5$，得 $3x + 2y = 5$。
  3. 求 $3x + 2y = 1$ 的特解：显然 $3 \times 1 + 2 \times (-1) = 1$，所以 $(1, -1)$ 是特解。
  4. $3x + 2y = 5$ 的特解：$(5 \times 1, 5 \times (-1)) = (5, -5)$。
  5. 通解： $$ \begin{cases} x = 5 + 2t \\ y = -5 - 3t \end{cases} \quad t \in \mathbb{Z} $$

示例2： 求解 $6x + 9y = 20$。

解法： 计算 $\gcd(6, 9) = 3$，$3 \nmid 20$，所以方程无整数解。

示例3： 求 $5x + 3y = 17$ 的所有正整数解。

解法：

  1. $\gcd(5, 3) = 1$，$1 \mid 17$，有解。
  2. 求 $5x + 3y = 1$ 的特解：$5 \times 2 + 3 \times (-3) = 1$，所以 $(2, -3)$ 是特解。
  3. $5x + 3y = 17$ 的特解：$(17 \times 2, 17 \times (-3)) = (34, -51)$。
  4. 通解： $$ \begin{cases} x = 34 + 3t \\ y = -51 - 5t \end{cases} \quad t \in \mathbb{Z} $$
  5. 要求 $x \gt 0$ 且 $y \gt 0$：
    • $34 + 3t \gt 0 \Rightarrow t \gt -34/3 \approx -11.33$，所以 $t \geq -11$
    • $-51 - 5t \gt 0 \Rightarrow -5t \gt 51 \Rightarrow t \lt -51/5 = -10.2$，所以 $t \leq -11$
  6. 同时满足 $t \geq -11$ 且 $t \leq -11$，所以 $t = -11$。
  7. 唯一正整数解：$x = 34 + 3 \times (-11) = 1$，$y = -51 - 5 \times (-11) = 4$。

代码示例 (C++)


#include <iostream>
#include <vector>
using namespace std;

// 扩展欧几里得算法
// 返回 gcd(a, b)，并通过引用参数返回一组满足 ax + by = gcd(a, b) 的整数解
long long extendedGcd(long long a, long long b, long long &x, long long &y) {
    if (b == 0) {
        x = 1;
        y = 0;
        return a;
    }
    long long x1, y1;
    long long g = extendedGcd(b, a % b, x1, y1);
    x = y1;
    y = x1 - (a / b) * y1;
    return g;
}

// 求解结果结构体
struct Solution {
    bool hasSolution;   // 是否有整数解
    long long x0, y0;   // 特解
    long long a1, b1;   // 约化后的系数
    bool allPairs;      // 是否所有整数对都是解 (a = b = c = 0)
};

// 正整数解结构体
struct PositiveSolution {
    long long x, y, t;
};

// 求解线性丢番图方程 ax + by = c
Solution solveDiophantine(long long a, long long b, long long c) {
    Solution sol{};
    sol.hasSolution = false;
    sol.allPairs = false;

    // 特殊情况处理
    if (a == 0 && b == 0) {
        if (c == 0) {
            sol.hasSolution = true;
            sol.allPairs = true;   // 所有整数对都是解
        }
        return sol;
    }

    // 使用扩展欧几里得算法（对绝对值求解）
    long long x, y;
    long long g = extendedGcd(a < 0 ? -a : a, b < 0 ? -b : b, x, y);

    // 检查解的存在性
    if (c % g != 0) {
        return sol;   // 无整数解
    }

    // 计算特解
    long long factor = c / g;
    long long x0 = x * factor;
    long long y0 = y * factor;

    // 调整符号
    if (a < 0) x0 = -x0;
    if (b < 0) y0 = -y0;

    // 约化系数
    long long a1 = a / g;
    long long b1 = b / g;

    sol.hasSolution = true;
    sol.x0 = x0;
    sol.y0 = y0;
    sol.a1 = a1;
    sol.b1 = b1;
    return sol;
}

// 寻找正整数解
vector<PositiveSolution> findPositiveSolutions(const Solution &sol, int maxT = 100) {
    vector<PositiveSolution> positiveSolutions;
    if (!sol.hasSolution) return positiveSolutions;

    long long x0 = sol.x0, y0 = sol.y0;
    long long a1 = sol.a1, b1 = sol.b1;

    // 通解：x = x0 + b1*t, y = y0 - a1*t
    // 寻找使 x>0 且 y>0 的 t
    for (int t = -maxT; t <= maxT; t++) {
        long long x = x0 + b1 * t;
        long long y = y0 - a1 * t;

        if (x > 0 && y > 0) {
            positiveSolutions.push_back({x, y, t});
        }
    }

    return positiveSolutions;
}

int main() {
    cout << "=== 线性丢番图方程求解 ===\n";

    // 示例1：21x + 14y = 35
    cout << "\n1. 方程：21x + 14y = 35\n";
    Solution sol1 = solveDiophantine(21, 14, 35);
    cout << "解存在: " << (sol1.hasSolution ? "true" : "false") << "\n";
    if (sol1.hasSolution) {
        cout << "特解: (" << sol1.x0 << ", " << sol1.y0 << ")\n";
        cout << "通解：\n";
        cout << "  x = " << sol1.x0 << " + " << sol1.b1 << "t\n";
        cout << "  y = " << sol1.y0 << " - " << sol1.a1 << "t\n";
        cout << "  其中 t 为任意整数\n";
    }

    // 示例2：6x + 9y = 20
    cout << "\n2. 方程：6x + 9y = 20\n";
    Solution sol2 = solveDiophantine(6, 9, 20);
    cout << "解存在: " << (sol2.hasSolution ? "true" : "false") << "\n";

    // 示例3：5x + 3y = 17 的正整数解
    cout << "\n3. 方程：5x + 3y = 17 的正整数解\n";
    Solution sol3 = solveDiophantine(5, 3, 17);
    cout << "解存在: " << (sol3.hasSolution ? "true" : "false") << "\n";
    if (sol3.hasSolution) {
        cout << "特解: (" << sol3.x0 << ", " << sol3.y0 << ")\n";

        vector<PositiveSolution> positiveSol = findPositiveSolutions(sol3, 20);
        cout << "正整数解（共" << positiveSol.size() << "个）:\n";
        for (size_t i = 0; i < positiveSol.size(); i++) {
            cout << "  t=" << positiveSol[i].t << ": (" << positiveSol[i].x << ", " << positiveSol[i].y << ")\n";
        }
    }

    // 示例4：更复杂的方程
    cout << "\n4. 方程：107x + 37y = 25\n";
    Solution sol4 = solveDiophantine(107, 37, 25);
    cout << "解存在: " << (sol4.hasSolution ? "true" : "false") << "\n";
    if (sol4.hasSolution) {
        cout << "特解: (" << sol4.x0 << ", " << sol4.y0 << ")\n";
        cout << "通解：\n";
        cout << "  x = " << sol4.x0 << " + " << sol4.b1 << "t\n";
        cout << "  y = " << sol4.y0 << " - " << sol4.a1 << "t\n";
        cout << "  其中 t 为任意整数\n";
    }

    // 验证解的正确性
    cout << "\n=== 验证解的正确性 ===\n";
    struct TestCase { long long a, b, c; };
    TestCase testCases[] = {
        { 21, 14, 35 },
        { 5, 3, 17 },
        { 107, 37, 25 }
    };
    for (const TestCase &test : testCases) {
        Solution sol = solveDiophantine(test.a, test.b, test.c);
        if (sol.hasSolution) {
            long long x = sol.x0, y = sol.y0;
            long long leftSide = test.a * x + test.b * y;
            cout << "验证：" << test.a << "×" << x << " + " << test.b << "×" << y
                 << " = " << leftSide << ", 目标值=" << test.c
                 << ", " << (leftSide == test.c ? "✓" : "✗") << "\n";
        }
    }

    return 0;
}

应用：

  1. 分配问题： 如"用5元和3元的邮票凑出17元"等问题。
  2. 同余方程求解： 方程 $ax \equiv c \pmod{b}$ 可化为 $ax + by = c$。
  3. 中国剩余定理： 构造解时需要求解线性丢番图方程。
  4. 密码学： RSA算法中计算私钥需要求解 $ed \equiv 1 \pmod{\varphi(n)}$。
  5. 计算机科学： 资源分配、调度等问题。

推广：多元线性丢番图方程

对于多元方程 $a_1x_1 + a_2x_2 + \cdots + a_nx_n = c$：

  1. 存在性条件： 有整数解当且仅当 $\gcd(a_1, a_2, \dots, a_n) \mid c$。
  2. 求解方法： 可化为多个二元方程求解，或使用矩阵方法（Smith标准形）。
  3. 通解结构： 若 $(x_1^0, x_2^0, \dots, x_n^0)$ 是一个特解，则通解可表示为特解加上齐次方程的通解。
