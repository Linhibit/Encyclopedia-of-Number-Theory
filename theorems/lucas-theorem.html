卢卡斯定理 (Lucas Theorem)

定理内容：

设 $p$ 为素数，$n$ 和 $m$ 为非负整数。将 $n$ 和 $m$ 表示为 $p$ 进制数： $$n = n_k p^k + n_{k-1} p^{k-1} + \cdots + n_1 p + n_0$$ $$m = m_k p^k + m_{k-1} p^{k-1} + \cdots + m_1 p + m_0$$ 其中 $0 \leq n_i, m_i \lt p$。

则有： $$\binom{n}{m} \equiv \prod_{i=0}^k \binom{n_i}{m_i} \pmod{p}$$

其中 $\binom{n}{m}$ 表示组合数 $C_n^m$。特别地，如果某个 $m_i \gt n_i$，则 $\binom{n_i}{m_i} = 0$，从而 $\binom{n}{m} \equiv 0 \pmod{p}$。

证明思路：

基于生成函数的证明：

  1. 考虑模 $p$ 意义下的二项式定理： $$(1+X)^n = \sum_{m=0}^n \binom{n}{m} X^m$$
  2. 由于 $p$ 是素数，有 $(1+X)^p \equiv 1 + X^p \pmod{p}$（这是因为 $p$ 整除 $\binom{p}{k}$ 对于 $0 \lt k \lt p$）。
  3. 将 $n$ 写成 $p$ 进制：$n = \sum_{i=0}^k n_i p^i$，则： $$ \begin{aligned} (1+X)^n &= \prod_{i=0}^k (1+X)^{n_i p^i} \\ &\equiv \prod_{i=0}^k (1+X^{p^i})^{n_i} \pmod{p} \end{aligned} $$
  4. 将右边展开，$X^m$ 的系数就是 $\prod_{i=0}^k \binom{n_i}{m_i} \pmod{p}$，其中 $m = \sum_{i=0}^k m_i p^i$。
  5. 比较两边 $X^m$ 的系数即得定理。

示例：

示例1： 计算 $\binom{100}{30} \bmod 13$。

解法：

  1. 将 100 和 30 转换为 13 进制：
    • $100 = 7 \times 13^1 + 9 \times 13^0$，所以 $n_1=7, n_0=9$
    • $30 = 2 \times 13^1 + 4 \times 13^0$，所以 $m_1=2, m_0=4$
  2. 根据卢卡斯定理： $$\binom{100}{30} \equiv \binom{7}{2} \times \binom{9}{4} \pmod{13}$$
  3. 计算：
    • $\binom{7}{2} = 21 \equiv 8 \pmod{13}$
    • $\binom{9}{4} = 126 \equiv 9 \pmod{13}$
  4. 所以 $\binom{100}{30} \equiv 8 \times 9 = 72 \equiv 7 \pmod{13}$。

示例2： 判断 $\binom{100}{45}$ 是否能被 7 整除。

解法：

  1. 将 100 和 45 转换为 7 进制：
    • $100 = 2 \times 7^2 + 0 \times 7^1 + 2 \times 7^0$，所以 $n_2=2, n_1=0, n_0=2$
    • $45 = 6 \times 7^1 + 3 \times 7^0$，所以 $m_1=6, m_0=3$（注意高位为0）
  2. 由于 $m_2=0 \leq n_2=2$，但 $m_1=6 \gt n_1=0$，所以 $\binom{n_1}{m_1} = \binom{0}{6} = 0$。
  3. 根据卢卡斯定理，$\binom{100}{45} \equiv 0 \pmod{7}$，即能被 7 整除。

代码示例 (C++)


#include <iostream>
#include <vector>
#include <algorithm>
#include <chrono>

using namespace std;
using ll = long long;

// 快速幂算法
ll modPow(ll base, ll exp, ll mod) {
    if (mod == 1) return 0;
    ll result = 1;
    base %= mod;
    while (exp > 0) {
        if (exp & 1) result = (result * base) % mod;
        exp >>= 1;
        base = (base * base) % mod;
    }
    return result;
}

// 预计算阶乘和阶乘的逆元（模p）
void precomputeFactorials(int n, int p, vector<ll>& fact, vector<ll>& invFact) {
    fact.assign(n + 1, 0);
    invFact.assign(n + 1, 0);

    fact[0] = 1;
    for (int i = 1; i <= n; i++) {
        fact[i] = (fact[i - 1] * i) % p;
    }

    // 费马小定理求逆元
    invFact[n] = modPow(fact[n], p - 2, p);
    for (int i = n - 1; i >= 0; i--) {
        invFact[i] = (invFact[i + 1] * (i + 1)) % p;
    }
}

// 使用预计算求组合数 C(n, m) mod p（n, m < p）
ll combModPrecomputed(int n, int m, const vector<ll>& fact, const vector<ll>& invFact, int p) {
    if (m < 0 || m > n) return 0;
    if (m == 0 || m == n) return 1;

    return (fact[n] * invFact[m] % p) * invFact[n - m] % p;
}

// 卢卡斯定理计算组合数 C(n, m) mod p
ll lucasTheorem(ll n, ll m, int p) {
    if (m == 0) return 1;

    // 将n, m转换为p进制
    vector<int> nDigits, mDigits;

    ll tempN = n, tempM = m;
    while (tempN > 0 || tempM > 0) {
        nDigits.push_back((int)(tempN % p));
        mDigits.push_back((int)(tempM % p));
        tempN /= p;
        tempM /= p;
    }

    // 确保两个数组长度相同
    int len = (int)max(nDigits.size(), mDigits.size());
    while ((int)nDigits.size() < len) nDigits.push_back(0);
    while ((int)mDigits.size() < len) mDigits.push_back(0);

    // 预计算阶乘（只需计算到p-1）
    vector<ll> fact, invFact;
    precomputeFactorials(p - 1, p, fact, invFact);

    // 应用卢卡斯定理
    ll result = 1;
    for (int i = 0; i < len; i++) {
        int ni = nDigits[i];
        int mi = mDigits[i];

        if (mi > ni) {
            return 0; // C(ni, mi) = 0
        }

        result = (result * combModPrecomputed(ni, mi, fact, invFact, p)) % p;
    }

    return result;
}

int main() {
    // 示例使用
    cout << "=== 卢卡斯定理示例 ===\n";

    // 示例1：C(100, 30) mod 13
    int p1 = 13;
    ll result1 = lucasTheorem(100, 30, p1);
    cout << "C(100, 30) mod " << p1 << " = " << result1 << '\n'; // 应为7

    // 示例2：C(100, 45) mod 7
    int p2 = 7;
    ll result2 = lucasTheorem(100, 45, p2);
    cout << "C(100, 45) mod " << p2 << " = " << result2 << '\n'; // 应为0

    // 示例3：验证小数字
    int p3 = 5;
    cout << "\n验证小数字（模" << p3 << "）:\n";
    for (int n = 0; n < 10; n++) {
        for (int m = 0; m <= n; m++) {
            // 直接计算（仅适用于n, m较小的情况）
            ll direct = 1;
            for (int i = 1; i <= m; i++) {
                direct = direct * (n - i + 1) / i;
            }
            direct = direct % p3;

            ll lucas = lucasTheorem(n, m, p3);

            if (direct != lucas) {
                cout << "矛盾：C(" << n << ", " << m << ") mod " << p3
                     << "，直接计算=" << direct << "，卢卡斯定理=" << lucas << '\n';
            }
        }
    }
    cout << "小数字验证完成！\n";

    // 性能测试
    cout << "\n=== 性能测试 ===\n";
    auto start = chrono::high_resolution_clock::now();
    ll bigResult = lucasTheorem(1000000, 500000, 10007);
    auto end = chrono::high_resolution_clock::now();
    auto ms = chrono::duration_cast<chrono::milliseconds>(end - start).count();
    cout << "卢卡斯定理: " << ms << "ms\n";
    cout << "C(1000000, 500000) mod 10007 = " << bigResult << '\n';

    return 0;
}

应用：

  1. 大组合数取模： 当 $n$ 和 $m$ 很大但模数 $p$ 是素数且不太大时，卢卡斯定理可以高效计算组合数模 $p$。
  2. 数论问题： 判断组合数是否能被素数整除。
  3. 组合数学： 研究二项式系数模素数的性质。
  4. 密码学： 在某些密码算法中用于计算大组合数。

📝 注意：

1. 卢卡斯定理要求模数 $p$ 是素数。对于合数模数，可以使用中国剩余定理分解为素数幂模数，然后对每个素数幂使用推广的卢卡斯定理。
2. 定理的一个重要推论：$\binom{n}{m}$ 被 $p$ 整除当且仅当在 $p$ 进制下，$m$ 的某一位大于 $n$ 的对应位。
3. 卢卡斯定理的计算复杂度为 $O(\log_p n)$，非常高效。
