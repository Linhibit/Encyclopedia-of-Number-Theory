佩尔方程 (Pell's Equation)

方程形式：

佩尔方程的标准形式为： $$x^2 - dy^2 = 1$$ 其中 $d$ 是一个非平方的正整数（即 $d$ 不是完全平方数）。

基本解： 设 $(x_1, y_1)$ 是所有正整数解中最小的解（称为基本解或最小解），则方程的所有正整数解 $(x_n, y_n)$ 可由下式生成： $$x_n + y_n\sqrt{d} = (x_1 + y_1\sqrt{d})^n, \quad n = 1, 2, 3, \dots$$

等价地，可以通过递推公式计算： $$ \begin{cases} x_{n+1} = x_1 x_n + d y_1 y_n \\ y_{n+1} = x_1 y_n + y_1 x_n \end{cases} $$

理论基础：

  1. 解的存在性： 对于非平方正整数 $d$，佩尔方程 $x^2 - dy^2 = 1$ 总有正整数解，且有无穷多组解。
  2. 基本解的唯一性： 存在唯一的最小正整数解 $(x_1, y_1)$，称为基本解。
  3. 解的结构： 所有解 $(x_n, y_n)$ 满足 $x_n + y_n\sqrt{d} = (x_1 + y_1\sqrt{d})^n$。
  4. 负佩尔方程： 方程 $x^2 - dy^2 = -1$ 不一定有解。如果有解，设其基本解为 $(x_0, y_0)$， 则 $x^2 - dy^2 = 1$ 的基本解为 $(x_0^2 + dy_0^2, 2x_0y_0)$。

求解方法：连分数法

$\sqrt{d}$ 的连分数展开是周期的。设其循环节长度为 $l$：

  1. 如果 $l$ 是偶数，则 $x^2 - dy^2 = 1$ 的基本解为连分数收敛的第 $l-1$ 个渐近分数 $p_{l-1}/q_{l-1}$。
  2. 如果 $l$ 是奇数，则 $x^2 - dy^2 = 1$ 的基本解为连分数收敛的第 $2l-1$ 个渐近分数 $p_{2l-1}/q_{2l-1}$。

连分数展开步骤：

  1. 设 $a_0 = \lfloor \sqrt{d} \rfloor$，$m_0 = 0$，$b_0 = 1$。
  2. 递推计算： $$ \begin{aligned} m_{k+1} &= b_k a_k - m_k \\ b_{k+1} &= \frac{d - m_{k+1}^2}{b_k} \\ a_{k+1} &= \left\lfloor \frac{a_0 + m_{k+1}}{b_{k+1}} \right\rfloor \end{aligned} $$
  3. 当出现 $a_k = 2a_0$ 时，连分数开始循环。

示例：

示例1： 求解 $x^2 - 2y^2 = 1$。

解法：

  1. $\sqrt{2} = 1.4142...$，连分数展开：$\sqrt{2} = [1;\overline{2}]$，循环节长度 $l=1$（奇数）。
  2. 由于 $l$ 是奇数，基本解为第 $2l-1=1$ 个渐近分数。
  3. 计算渐近分数：
    • $p_0 = 1, q_0 = 1$
    • $p_1 = 1 \times 2 + 1 = 3, q_1 = 2$
  4. 验证：$3^2 - 2 \times 2^2 = 9 - 8 = 1$，所以基本解为 $(3, 2)$。
  5. 通解：$x_n + y_n\sqrt{2} = (3 + 2\sqrt{2})^n$。

示例2： 求解 $x^2 - 7y^2 = 1$。

解法：

  1. $\sqrt{7} = 2.6457...$，连分数展开：$\sqrt{7} = [2; \overline{1, 1, 1, 4}]$，循环节长度 $l=4$（偶数）。
  2. 由于 $l$ 是偶数，基本解为第 $l-1=3$ 个渐近分数。
  3. 计算渐近分数：
    • $a_0 = 2, a_1 = 1, a_2 = 1, a_3 = 1, a_4 = 4$
    • $p_0 = 2, q_0 = 1$
    • $p_1 = 2 \times 1 + 1 = 3, q_1 = 1$
    • $p_2 = 3 \times 1 + 2 = 5, q_2 = 1 \times 1 + 1 = 2$
    • $p_3 = 5 \times 1 + 3 = 8, q_3 = 2 \times 1 + 1 = 3$
  4. 验证：$8^2 - 7 \times 3^2 = 64 - 63 = 1$，所以基本解为 $(8, 3)$。

代码示例 (C++)


// 计算 d 的平方根连分数展开，并求解佩尔方程 x² - dy² = 1
#include <iostream>
#include <vector>
#include <map>
#include <string>
#include <cmath>
using namespace std;
using ll = long long;

// 连分数展开结果：a0 与循环部分
struct ContinuedFraction {
    ll a0;
    vector<ll> periodic; // 循环部分
};

// 计算 d 的平方根连分数展开
ContinuedFraction continuedFractionSqrt(ll d) {
    ll a0 = (ll)floor(sqrt((double)d));
    vector<ll> terms;
    terms.push_back(a0);
    map<string, ll> seen; // 记录 (m,b,a) 是否出现过
    ll m = 0, b = 1, a = a0;

    while (true) {
        m = b * a - m;
        b = (d - m * m) / b;
        a = (a0 + m) / b;

        string key = to_string(m) + "," + to_string(b) + "," + to_string(a);
        if (seen.count(key)) break;

        seen[key] = (ll)terms.size();
        terms.push_back(a);
    }

    ContinuedFraction cf;
    cf.a0 = a0;
    for (size_t i = 1; i < terms.size(); ++i) cf.periodic.push_back(terms[i]);
    return cf;
}

// 计算连分数的渐近分数
void convergents(const vector<ll>& terms, vector<ll>& p, vector<ll>& q) {
    p.clear();
    q.clear();
    p.push_back(terms[0]);
    p.push_back(terms[0] * terms[1] + 1);
    q.push_back(1);
    q.push_back(terms[1]);

    for (size_t i = 2; i < terms.size(); ++i) {
        p.push_back(terms[i] * p[i - 1] + p[i - 2]);
        q.push_back(terms[i] * q[i - 1] + q[i - 2]);
    }
}

// 佩尔方程的一个解
struct PellSolution {
    ll x;
    ll y;
    ll index;
};

// 求解佩尔方程 x² - dy² = 1 的基本解
PellSolution solvePellBasic(ll d) {
    ContinuedFraction cf = continuedFractionSqrt(d);
    ll periodicLen = (ll)cf.periodic.size();

    // 扩展循环部分以获得足够的项
    vector<ll> extendedTerms;
    extendedTerms.push_back(cf.a0);
    for (ll i = 0; i < periodicLen * 2; ++i) {
        extendedTerms.push_back(cf.periodic[i % periodicLen]);
    }

    vector<ll> p, q;
    convergents(extendedTerms, p, q);

    // 根据循环节长度确定基本解的位置
    ll index;
    if (periodicLen % 2 == 0) {
        // 偶数循环节，基本解在 l-1 处
        index = periodicLen - 1;
    } else {
        // 奇数循环节，基本解在 2l-1 处
        index = 2 * periodicLen - 1;
    }

    PellSolution sol;
    sol.x = p[index];
    sol.y = q[index];
    sol.index = index;
    return sol;
}

// 生成佩尔方程的第 n 个解
PellSolution pellSolution(ll d, ll n) {
    PellSolution basic = solvePellBasic(d);
    ll x1 = basic.x;
    ll y1 = basic.y;

    // 使用快速幂计算 (x1 + y1√d)^n
    // 中间结果可能超出 64 位，用 __int128 避免溢出
    __int128 x = 1, y = 0;
    __int128 xPow = x1, yPow = y1;
    __int128 dd = d;
    __int128 e = n;

    while (e > 0) {
        if (e % 2 == 1) {
            // 乘以当前幂
            __int128 newX = x * xPow + dd * y * yPow;
            __int128 newY = x * yPow + y * xPow;
            x = newX;
            y = newY;
        }
        // 平方当前幂
        __int128 newXPow = xPow * xPow + dd * yPow * yPow;
        __int128 newYPow = 2 * xPow * yPow;
        xPow = newXPow;
        yPow = newYPow;

        e >>= 1;
    }

    PellSolution sol;
    sol.x = (ll)x;
    sol.y = (ll)y;
    sol.index = n;
    return sol;
}

int main() {
    cout << "=== 佩尔方程求解 ===" << '\n';

    // 示例1：x² - 2y² = 1
    cout << "\n1. x² - 2y² = 1" << '\n';
    PellSolution sol1 = solvePellBasic(2);
    cout << "基本解: (" << sol1.x << ", " << sol1.y << ")" << '\n';
    cout << "验证: " << sol1.x << "² - 2×" << sol1.y << "² = "
         << (sol1.x * sol1.x - 2 * sol1.y * sol1.y) << '\n';

    // 生成前5个解
    cout << "前5个解:" << '\n';
    for (ll n = 1; n <= 5; ++n) {
        PellSolution sol = pellSolution(2, n);
        cout << "n=" << n << ": (" << sol.x << ", " << sol.y << ")" << '\n';
    }

    // 示例2：x² - 7y² = 1
    cout << "\n2. x² - 7y² = 1" << '\n';
    PellSolution sol2 = solvePellBasic(7);
    cout << "基本解: (" << sol2.x << ", " << sol2.y << ")" << '\n';
    cout << "验证: " << sol2.x << "² - 7×" << sol2.y << "² = "
         << (sol2.x * sol2.x - 7 * sol2.y * sol2.y) << '\n';

    // 示例3：x² - 13y² = 1
    cout << "\n3. x² - 13y² = 1" << '\n';
    PellSolution sol3 = solvePellBasic(13);
    cout << "基本解: (" << sol3.x << ", " << sol3.y << ")" << '\n';
    cout << "验证: " << sol3.x << "² - 13×" << sol3.y << "² = "
         << (sol3.x * sol3.x - 13 * sol3.y * sol3.y) << '\n';

    // 验证连分数展开
    cout << "\n=== 连分数展开验证 ===" << '\n';
    ll testD[] = {2, 3, 5, 6, 7, 8, 10, 11, 12, 13};
    for (ll d : testD) {
        ContinuedFraction cf = continuedFractionSqrt(d);
        cout << "√" << d << " = [" << cf.a0 << "; ";
        for (size_t i = 0; i < cf.periodic.size(); ++i) {
            if (i > 0) cout << ", ";
            cout << cf.periodic[i];
        }
        cout << "]" << '\n';
        cout << "  循环节长度: " << cf.periodic.size() << '\n';
    }

    return 0;
}

应用：

  1. 代数数论： 研究二次域的单位群。
  2. 丢番图逼近： 提供 $\sqrt{d}$ 的最佳有理逼近。
  3. 组合数学： 某些递推关系的通项公式。
  4. 密码学： 某些公钥密码系统的构造。
  5. 数学史： 古代数学问题（如"牛群问题"）的求解。

📝 历史背景：

佩尔方程得名于英国数学家约翰·佩尔（John Pell，1611-1685），但实际上他对这个方程的贡献很小。 这个命名是欧拉的一个错误造成的。

佩尔方程在古代就被研究过。阿基米德的"牛群问题"涉及佩尔方程 $x^2 - 4729494y^2 = 1$， 但直到20世纪才完全解决。

印度数学家婆罗摩笈多（Brahmagupta，598-668）发现了佩尔方程的"再生"性质： 如果 $(x_1, y_1)$ 和 $(x_2, y_2)$ 是解，那么 $(x_1x_2 + dy_1y_2, x_1y_2 + x_2y_1)$ 也是解。
