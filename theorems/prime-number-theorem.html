素数定理 (Prime Number Theorem)

定理内容：

设 $\pi(x)$ 表示不超过 $x$ 的素数的个数，即： $$\pi(x) = \#\{p \leq x : p \text{ 是素数}\}$$

当 $x$ 趋近于无穷大时，$\pi(x)$ 的渐近行为为： $$\pi(x) \sim \frac{x}{\ln x}$$

其中符号"$\sim$"表示渐近等价，即： $$\lim_{x \to \infty} \frac{\pi(x)}{x / \ln x} = 1$$

更精确的形式（由高斯提出）： $$\pi(x) \sim \operatorname{Li}(x) = \int_2^x \frac{dt}{\ln t}$$ 其中 $\operatorname{Li}(x)$ 是对数积分函数。

定理的直观解释：

  1. 概率解释： 在 $1, 2, \dots, n$ 中随机选择一个数，它是素数的概率约为 $\frac{1}{\ln n}$。
  2. 素数间隔： 在 $n$ 附近的素数平均间隔约为 $\ln n$。
  3. 素数密度： 素数的"密度"在 $x$ 处约为 $\frac{1}{\ln x}$，随着 $x$ 增大而减小。
  4. 渐近意义： 当 $x$ 非常大时，$\pi(x)$ 与 $x/\ln x$ 的比值趋近于 1，但两者的差可能很大。

更精确的估计： $$\pi(x) = \frac{x}{\ln x} + \frac{x}{(\ln x)^2} + \frac{2!x}{(\ln x)^3} + \cdots + \frac{(k-1)!x}{(\ln x)^k} + O\left(\frac{x}{(\ln x)^{k+1}}\right)$$ 这是一个渐近级数，对固定的 $k$，当 $x$ 足够大时误差项趋于0。

数据对比：$\pi(x)$ 的实际值与近似值

从表中可以看出：

  • 随着 $x$ 增大，比值 $\pi(x) / (x/\ln x)$ 逐渐趋近于 1
  • $\operatorname{Li}(x)$ 比 $x/\ln x$ 更接近 $\pi(x)$ 的实际值
  • 即使对于 $10^{15}$ 这样大的数，近似仍有约 3% 的误差

代码示例 (C++)

注意： 计算 $\pi(x)$ 的确切值对于大 $x$ 计算量很大，上述代码的筛法只能处理中等大小的 $x$。 实际应用中，对于非常大的 $x$，有专门的算法（如Meissel-Lehmer算法）可以高效计算 $\pi(x)$。 


#include <iostream>
#include <vector>
#include <cmath>
#include <iomanip>
#include <string>
using namespace std;

// 计算不超过n的素数个数（埃氏筛法）
long long primePi(long long n) {
    if (n < 2) return 0;
    
    long long limit = (long long)floor(sqrt((double)n));
    vector<bool> isPrime(n + 1, true);
    isPrime[0] = false;
    isPrime[1] = false;
    
    // 埃氏筛法
    for (long long p = 2; p <= limit; p++) {
        if (isPrime[p]) {
            for (long long multiple = p * p; multiple <= n; multiple += p) {
                isPrime[multiple] = false;
            }
        }
    }
    
    // 计数
    long long count = 0;
    for (long long i = 2; i <= n; i++) {
        if (isPrime[i]) count++;
    }
    
    return count;
}

// 计算x/ln(x)
double xOverLnX(double x) {
    return x / log(x);
}

// 计算对数积分Li(x)的近似值（数值积分）
double logarithmicIntegral(double x) {
    // 使用梯形法则进行数值积分
    int n = 10000; // 分割数
    double step = (x - 2) / n;
    double sum = 0;
    
    // 梯形法则：∫f(x)dx ≈ (h/2)[f(x0)+2f(x1)+...+2f(x_{n-1})+f(xn)]
    for (int i = 0; i <= n; i++) {
        double t = 2 + i * step;
        double weight = (i == 0 || i == n) ? 1 : 2;
        sum += weight * (1 / log(t));
    }
    
    return (step / 2) * sum;
}

// 判断一个数是否为素数
bool isPrime(long long num) {
    if (num < 2) return false;
    if (num % 2 == 0) return num == 2;
    long long limit = (long long)floor(sqrt((double)num));
    for (long long i = 3; i <= limit; i += 2) {
        if (num % i == 0) return false;
    }
    return true;
}

// 获取第n个素数
long long nthPrime(long long n) {
    if (n < 1) return -1;
    
    long long count = 0;
    long long num = 2;
    
    while (true) {
        if (isPrime(num)) {
            count++;
            if (count == n) return num;
        }
        num++;
    }
}

// 验证素数定理的近似程度
void verifyPrimeNumberTheorem(long long limit) {
    cout << "验证素数定理：\n";
    cout << "x\tπ(x)\tx/ln(x)\tLi(x)\tπ(x)/(x/ln(x))\n";
    cout << string(60, '-') << "\n";
    
    long long testPoints[] = {100, 1000, 10000, 100000, 1000000};
    
    for (long long x : testPoints) {
        if (x > limit) break;
        
        long long piX = primePi(x);
        double xOverLnXVal = xOverLnX((double)x);
        double liX = logarithmicIntegral((double)x);
        double ratio = piX / xOverLnXVal;
        
        cout << x << "\t" << piX << "\t"
             << fixed << setprecision(2) << xOverLnXVal << "\t"
             << setprecision(2) << liX << "\t"
             << setprecision(4) << ratio << "\n";
    }
}

int main() {
    // 示例使用
    cout << "=== 素数定理验证 ===\n";
    verifyPrimeNumberTheorem(1000000);
    
    // 计算误差
    cout << "\n=== 计算不同近似公式的绝对误差 ===\n";
    long long x = 1000000;
    long long piX = primePi(x);
    double approx1 = xOverLnX((double)x);
    double approx2 = logarithmicIntegral((double)x);
    
    cout << "π(" << x << ") = " << piX << "\n";
    cout << "x/ln(x) = " << fixed << setprecision(2) << approx1
         << ", 绝对误差 = " << setprecision(2) << fabs(piX - approx1) << "\n";
    cout << "Li(x) = " << setprecision(2) << approx2
         << ", 绝对误差 = " << setprecision(2) << fabs(piX - approx2) << "\n";
    cout << "x/ln(x)的相对误差: " << setprecision(2) << (fabs(piX - approx1) / piX * 100) << "%\n";
    cout << "Li(x)的相对误差: " << setprecision(2) << (fabs(piX - approx2) / piX * 100) << "%\n";
    
    // 素数定理的一个推论：第n个素数的渐近公式
    cout << "\n=== 第n个素数的渐近公式 ===\n";
    cout << "素数定理的一个推论：第n个素数p_n ~ n ln n\n";
    cout << "验证：\n";
    
    long long testNs[] = {1, 10, 100, 1000, 10000};
    cout << "n\tp_n(实际)\tn ln n\t\tp_n/(n ln n)\n";
    
    for (long long n : testNs) {
        long long pn = nthPrime(n);
        double approx = n * log((double)n);
        cout << n << "\t" << pn << "\t\t"
             << fixed << setprecision(2) << approx << "\t"
             << setprecision(4) << (pn / approx) << "\n";
    }
    
    return 0;
}

📝 历史背景：

猜想阶段： 1792年，15岁的高斯在研究素数表时，猜测素数定理的形式。 勒让德在1798年也独立提出了类似的猜想。

初步进展： 切比雪夫在1852年证明了存在常数 $c_1, c_2$ 使得对于充分大的 $x$： $$c_1 \frac{x}{\ln x} \lt \pi(x) \lt c_2 \frac{x}{\ln x}$$ 他取 $c_1 = 0.92129, c_2 = 1.10555$。

证明： 1896年，阿达马（Jacques Hadamard）和瓦莱·普桑（Charles de la Vallée Poussin）分别独立证明了素数定理。 他们的证明都使用了黎曼zeta函数的性质，属于解析数论的方法。

初等证明： 1949年，塞尔伯格（Atle Selberg）和爱尔特希（Paul Erdős）分别给出了不依赖复分析的"初等证明"， 虽然证明仍然相当复杂。

推论：

  1. 第 $n$ 个素数： 设 $p_n$ 表示第 $n$ 个素数，则： $$p_n \sim n \ln n$$ 更精确地：$p_n = n \ln n + n \ln \ln n + O(n)$。
  2. 素数间隔： 存在任意长的连续合数段，即对于任意 $k$，存在 $n$ 使得 $n+1, n+2, \dots, n+k$ 都是合数。
  3. 伯特兰公设： 对于任意 $n \gt 1$，存在素数 $p$ 满足 $n \lt p \lt 2n$。 这是切比雪夫证明的，比素数定理弱但不需要解析方法。
  4. 平均间隔： 在 $x$ 附近的素数平均间隔约为 $\ln x$。

与黎曼猜想的关系：

黎曼猜想是数论中最著名的未解决问题。如果黎曼猜想成立，那么素数定理的误差项可以大幅改进： $$\pi(x) = \operatorname{Li}(x) + O(\sqrt{x} \ln x)$$

目前无条件的最佳误差界（Vinogradov-Korobov，1958）： $$\pi(x) = \operatorname{Li}(x) + O\left(x \exp(-c(\ln x)^{3/5}(\ln \ln x)^{-1/5})\right)$$

如果黎曼猜想成立，我们就能更精确地了解素数的分布规律。
