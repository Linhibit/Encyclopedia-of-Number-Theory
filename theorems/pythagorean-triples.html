勾股数公式 (Pythagorean Triple Formulas)

勾股数定义：

满足 $a^2 + b^2 = c^2$ 的正整数三元组 $(a, b, c)$ 称为勾股数或毕达哥拉斯三元组。

如果 $\gcd(a, b, c) = 1$，则称 $(a, b, c)$ 为本原勾股数。

本原勾股数生成公式：

所有本原勾股数 $(a, b, c)$（其中 $a$ 是奇数，$b$ 是偶数）都可以由以下公式生成： $$ \begin{cases} a = m^2 - n^2 \\ b = 2mn \\ c = m^2 + n^2 \end{cases} $$ 其中 $m$ 和 $n$ 是满足以下条件的正整数：

  1. $m \gt n \gt 0$
  2. $\gcd(m, n) = 1$（即 $m$ 与 $n$ 互质）
  3. $m$ 和 $n$ 一奇一偶（即 $m + n$ 是奇数）

所有勾股数： 一般的勾股数可以通过将本原勾股数乘以一个正整数 $k$ 得到： $$(ka, kb, kc) = (k(m^2 - n^2), 2kmn, k(m^2 + n^2))$$

证明思路：

必要性（公式生成的是本原勾股数）：

设 $m \gt n \gt 0$，$\gcd(m, n) = 1$，且 $m$ 和 $n$ 一奇一偶。令： $$a = m^2 - n^2, \quad b = 2mn, \quad c = m^2 + n^2$$ 验证： $$a^2 + b^2 = (m^2 - n^2)^2 + (2mn)^2 = m^4 - 2m^2n^2 + n^4 + 4m^2n^2 = m^4 + 2m^2n^2 + n^4 = (m^2 + n^2)^2 = c^2$$ 再证 $\gcd(a, b, c) = 1$：

  • 由于 $m$ 和 $n$ 一奇一偶，所以 $a$ 是奇数，$b$ 是偶数。
  • 假设素数 $p$ 整除 $a$ 和 $b$，则 $p$ 整除 $b^2 = 4m^2n^2$ 和 $c^2 = a^2 + b^2$，从而 $p$ 整除 $c$。
  • 但 $c + a = 2m^2$，$c - a = 2n^2$，所以 $p$ 整除 $2m^2$ 和 $2n^2$。
  • 由于 $p$ 是奇数（因为 $a$ 是奇数），$p$ 整除 $m^2$ 和 $n^2$，从而 $p$ 整除 $m$ 和 $n$，与 $\gcd(m, n) = 1$ 矛盾。
因此 $\gcd(a, b, c) = 1$，确实是本原勾股数。

充分性（所有本原勾股数都可以这样表示）：

设 $(a, b, c)$ 是本原勾股数，不妨设 $a$ 是奇数，$b$ 是偶数（否则交换 $a, b$）。 则方程 $a^2 + b^2 = c^2$ 可以改写为： $$\left(\frac{b}{2}\right)^2 = \frac{c^2 - a^2}{4} = \frac{c+a}{2} \cdot \frac{c-a}{2}$$ 可以证明 $\frac{c+a}{2}$ 和 $\frac{c-a}{2}$ 互质且都是完全平方数， 设 $\frac{c+a}{2} = m^2$，$\frac{c-a}{2} = n^2$，则： $$a = m^2 - n^2, \quad b = 2mn, \quad c = m^2 + n^2$$ 且 $m \gt n \gt 0$，$\gcd(m, n) = 1$，$m$ 和 $n$ 一奇一偶。

示例：

示例1： 生成一些本原勾股数。

$m$ $n$ $a=m^2-n^2$ $b=2mn$ $c=m^2+n^2$ 验证 $a^2+b^2=c^2$
2 1 3 4 5 $3^2+4^2=9+16=25=5^2$
3 2 5 12 13 $5^2+12^2=25+144=169=13^2$
4 1 15 8 17 $15^2+8^2=225+64=289=17^2$
4 3 7 24 25 $7^2+24^2=49+576=625=25^2$
5 2 21 20 29 $21^2+20^2=441+400=841=29^2$

示例2： 判断 $(9, 12, 15)$ 是否是本原勾股数。

解法： $\gcd(9, 12, 15) = 3 \neq 1$，所以不是本原勾股数。 但它可以由本原勾股数 $(3, 4, 5)$ 乘以 $3$ 得到。

示例3： 求面积为 $84$ 的直角三角形的边长（边长为整数）。

解法： 设直角边为 $a, b$，斜边为 $c$，则面积 $S = \frac{1}{2}ab = 84$，即 $ab = 168$。 同时 $a^2 + b^2 = c^2$。枚举可能的 $a, b$：

  • 若 $(a, b) = (12, 14)$，则 $ab=168$，但 $12^2+14^2=144+196=340$ 不是完全平方数。
  • 若 $(a, b) = (8, 21)$，则 $8^2+21^2=64+441=505$ 不是完全平方数。
  • 若 $(a, b) = (7, 24)$，则 $7^2+24^2=49+576=625=25^2$，符合条件。
所以边长为 $(7, 24, 25)$，面积为 $\frac{1}{2} \times 7 \times 24 = 84$。

代码示例 (C++)


#include <iostream>
#include <vector>
#include <algorithm>
#include <cmath>

using namespace std;
using ll = long long;

// 最大公约数
ll gcd(ll a, ll b) {
    return b == 0 ? a : gcd(b, a % b);
}

// 验证勾股数
bool verifyTriple(ll a, ll b, ll c) {
    return a * a + b * b == c * c;
}

// 判断是否为本原勾股数
bool isPrimitiveTriple(ll a, ll b, ll c) {
    return gcd(gcd(a, b), c) == 1;
}

// 生成本原勾股数
vector<vector<ll>> generatePrimitiveTriples(ll limit) {
    vector<vector<ll>> triples;

    // m和n的范围：m^2 + n^2 <= limit
    ll maxM = (ll)floor(sqrt((double)limit));

    for (ll m = 2; m <= maxM; m++) {
        ll maxN = min(m - 1, (ll)floor(sqrt((double)(limit - m * m))));
        for (ll n = 1; n <= maxN; n++) {
            // 检查条件：m > n, gcd(m, n) = 1, m和n一奇一偶
            if (gcd(m, n) == 1 && (m % 2 != n % 2)) {
                ll a = m * m - n * n;
                ll b = 2 * m * n;
                ll c = m * m + n * n;

                if (c <= limit) {
                    // 确保a是正数（实际上m>n保证a>0）
                    triples.push_back({a, b, c});
                }
            }
        }
    }

    // 按c从小到大排序
    sort(triples.begin(), triples.end(),
         [](const vector<ll>& t1, const vector<ll>& t2) { return t1[2] < t2[2]; });

    return triples;
}

// 生成所有勾股数（包括非本原）
struct Triple {
    ll a, b, c;
    bool primitive;
    ll multiple;
};

vector<Triple> generateAllTriples(ll limit, ll maxK) {
    vector<vector<ll>> primitives = generatePrimitiveTriples(limit);
    vector<Triple> allTriples;

    // 添加本原勾股数
    for (auto triple : primitives) {
        allTriples.push_back({triple[0], triple[1], triple[2], true, 1});
    }

    // 添加非本原勾股数
    for (auto triple : primitives) {
        ll a = triple[0], b = triple[1], c = triple[2];
        for (ll k = 2; k <= maxK; k++) {
            ll ka = a * k;
            ll kb = b * k;
            ll kc = c * k;

            if (kc <= limit) {
                allTriples.push_back({ka, kb, kc, false, k});
            } else {
                break;
            }
        }
    }

    // 按斜边c排序
    sort(allTriples.begin(), allTriples.end(),
         [](const Triple& t1, const Triple& t2) { return t1.c < t2.c; });

    return allTriples;
}

// 查找特定条件的勾股数
vector<vector<ll>> findTripleByArea(ll area) {
    // 面积 S = ab/2，所以 ab = 2S
    ll targetProduct = 2 * area;
    vector<vector<ll>> candidates;

    // 枚举a和b
    for (ll a = 1; a * a <= targetProduct; a++) {
        if (targetProduct % a == 0) {
            ll b = targetProduct / a;
            ll c2 = a * a + b * b;
            double c = sqrt((double)c2);

            if (abs(c - round(c)) < 1e-10) {
                candidates.push_back({a, b, (ll)round(c)});
            }
        }
    }

    return candidates;
}

// 计算勾股数的其他性质
struct Properties {
    ll perimeter;
    double area;
    double inradius;
    double circumradius;
};

Properties tripleProperties(ll a, ll b, ll c) {
    ll perimeter = a + b + c;
    double area = 0.5 * a * b;
    double inradius = area / (0.5 * perimeter); // 内切圆半径
    double circumradius = c / 2.0; // 外接圆半径

    return {perimeter, area, inradius, circumradius};
}

int main() {
    // 示例使用
    cout << "=== 生成本原勾股数（c ≤ 50） ===" << '\n';
    vector<vector<ll>> primitives = generatePrimitiveTriples(50);
    cout << "共找到 " << primitives.size() << " 个本原勾股数:" << '\n';
    for (auto triple : primitives) {
        ll a = triple[0], b = triple[1], c = triple[2];
        cout << "(" << a << ", " << b << ", " << c << ")，验证："
             << a << "²+" << b << "²=" << a * a << "+" << b * b << "=" << c << "²，"
             << (verifyTriple(a, b, c) ? "✓" : "✗") << '\n';
    }

    // 生成所有勾股数（c ≤ 50）
    cout << "\n=== 生成所有勾股数（c ≤ 50） ===" << '\n';
    vector<Triple> allTriples = generateAllTriples(50, 5);
    cout << "共找到 " << allTriples.size() << " 个勾股数（包括非本原）:" << '\n';
    for (auto t : allTriples) {
        cout << "(" << t.a << ", " << t.b << ", " << t.c << ") "
             << (t.primitive ? "[本原]" : "[非本原]") << '\n';
    }

    // 查找特定条件的勾股数
    cout << "\n=== 查找面积为84的直角三角形 ===" << '\n';
    vector<vector<ll>> area84 = findTripleByArea(84);
    cout << "面积为84的勾股数:" << '\n';
    for (auto triple : area84) {
        ll a = triple[0], b = triple[1], c = triple[2];
        cout << "(" << a << ", " << b << ", " << c << ")，面积="
             << 0.5 * a * b << "，"
             << (verifyTriple(a, b, c) ? "有效" : "无效") << '\n';
    }

    // 验证勾股数公式
    cout << "\n=== 验证公式正确性 ===" << '\n';
    cout << "测试m=5, n=2:" << '\n';
    ll m = 5, n = 2;
    ll a = m * m - n * n;
    ll b = 2 * m * n;
    ll c = m * m + n * n;
    cout << "a = " << m << "² - " << n << "² = " << a << '\n';
    cout << "b = 2×" << m << "×" << n << " = " << b << '\n';
    cout << "c = " << m << "² + " << n << "² = " << c << '\n';
    cout << "验证: " << a << "² + " << b << "² = " << a * a << " + " << b * b
         << " = " << a * a + b * b << ", " << c << "² = " << c * c << '\n';
    cout << "是否勾股数: " << (verifyTriple(a, b, c) ? "是" : "否") << '\n';
    cout << "是否本原: " << (isPrimitiveTriple(a, b, c) ? "是" : "否") << '\n';

    // 计算勾股数的其他性质
    cout << "\n=== 勾股数的性质 ===" << '\n';
    ll x = 3, y = 4, z = 5;
    Properties props = tripleProperties(x, y, z);
    cout << "勾股数 (" << x << ", " << y << ", " << z << ") 的性质:" << '\n';
    cout << "周长: " << props.perimeter << '\n';
    cout << "面积: " << props.area << '\n';
    cout << "内切圆半径: " << props.inradius << '\n';
    cout << "外接圆半径: " << props.circumradius << '\n';

    return 0;
}

应用：

  1. 几何学： 直角三角形边长的整数解。
  2. 数论： 平方和问题，费马平方和定理等。
  3. 密码学： 某些加密算法中用到勾股数。
  4. 计算机图形学： 生成整数坐标的直角三角形。
  5. 数学竞赛： 常见的数论问题。

相关推广：

  • 费马大定理： 方程 $x^n + y^n = z^n$ 在 $n \gt 2$ 时没有正整数解。 这是勾股数方程的推广，但结论完全不同。
  • 椭圆曲线： 勾股数方程可以转化为椭圆曲线，与现代数论有深刻联系。

📝 历史背景：

勾股数在古代文明中就被发现。巴比伦泥板普林顿322（约公元前1800年）列出了15组勾股数。

欧几里得在《几何原本》中给出了勾股数公式的证明。但现代形式的公式通常归功于古希腊数学家丢番图。

勾股数公式是数论中为数不多的能给出所有解的完全参数化公式之一， 这使它成为研究不定方程的典范。
